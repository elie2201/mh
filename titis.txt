function trimToEmpty(str) {

function trimToDefault(str, deflt) {
  return str === 0 ? '0' : ((str || (deflt ? deflt : '')) + '').trim();
}
function logging(...any) {
  const debug = true;
  if (debug) {
    console.groupCollapsed(new Date(), ...any);
    console.log(Error().stack);
    console.groupEnd();
  }
}
function getStorageToObject(keyName, objDefault) {
  let obj = getStorage(keyName);
  let bCheckNewProp = true;
  if (isNullOrUndefined(obj)) {
    obj = JSON.stringify(objDefault);
    setStorage(keyName, obj);
    bCheckNewProp = false;
  }
  obj = JSON.parse(obj);
  if (bCheckNewProp) {
    if (assignMissingDefault(obj, objDefault)) {
      setStorage(keyName, JSON.stringify(obj));
    }
  }
  return obj;
  function getStorage(name) {
    // check if the web browser support HTML5 storage
    if ('localStorage' in window && !isNullOrUndefined(window.localStorage)) {
      return window.localStorage.getItem(name);
    }
  }
  function setStorage(name, value) {
    // check if the web browser support HTML5 storage
    if ('localStorage' in window && !isNullOrUndefined(window.localStorage)) {
      window.localStorage.setItem(name, value);
    }
  }
  function isNullOrUndefined(obj) {
    return (
      obj === null || obj === undefined || obj === 'null' || obj === 'undefined'
    );
  }
  function assignMissingDefault(target, source) {
    let isResave = false;
    for (const key in source) {
      if (
        Object.prototype.hasOwnProperty.call(source, key) &&
        !Object.prototype.hasOwnProperty.call(target, key)
      ) {
        const element = source[key];
        target[key] = element;
        isResave = true;
      }
    }
    return isResave;
  }
}

/** Remove any comma in number expression and then parse to integer
 * return 0 if argument is null or isNaN(after remove comma)
 */
function parseQuantity(qty) {
  const strQty = (qty || 0) + '';
  const rtn = parseInt(strQty.replaceAll(',', ''));
  return Number.isNaN(rtn) ? 0 : rtn;
}
/**
 * Query item quantity by http get and then execute callback
 * @param {string} itemType
 * @param {Function} callback
 */
function getQuantityByUrlThen(itemType, callback, isAlwaysCallback) {
  const url = MOUSEHUNT_BASE_URL + 'item.php?item_type=' + itemType;
  $.get(url, (data) => {
    const key = 'You Own: ';
    const text = data.split(key)[1];
    const key2 = '</div>';
    const endIndex = text.indexOf(key2);
    const qtyText = text.substring(0, endIndex).trim();
    logging(qtyText);
    const qty = parseQuantity(qtyText.split('/')[0]);
    if (!isAlwaysCallback && (Number.isNaN(qty) || qty < 1)) return;
    if (callback) callback(qty);
  });
}

const tribalIslesTyps = ['derr_dunes', 'nerg_plains', 'elub_shore'];
if (tribalIslesTyps.indexOf(getEnvironmentType()) > -1) tribalIsles();
function tribalIsles() {
  const isProd = false;
  const targetEnvironmentType = 'tribalIsles';
  const environmentType = getEnvironmentType();
  if (tribalIslesTyps.indexOf(environmentType) < 0) return;
  logging('tribalIsles_0.0.0.2.1');
  // my own hard-coded constants
  const derr = 'derr_dunes';
  const nerg = 'nerg_plains';
  const elub = 'elub_shore';
  const gouda = 'gouda';
  const boss = 'boss';
  const delicious = 'Delicious';
  const savoury = 'Savoury';
  const seashell = 'Seashell';
  const items = {
    Delicious: 'delicious_stone_craft_item',
    Savoury: 'savoury_vegetables_craft_item',
    Seashell: 'seashell_craft_item'
  };
  /* const tribalBases = objBestTrap.base.combo.concat([
    'Tiki Base',
    'Tribal Base'
  ]); */
  /** localStorage settings */
  // prettier-ignore
  const defaultSettings = {
    traps: {
      derr_dunes: {
        gouda: ['Gouda Cheese', 'best.weapon.physical', 'Tribal Base,Aqua Base', 'Power Charm', false, 'none', false],
        boss: ['Crunchy Cheese', 'best.weapon.physical', 'Tribal Base,Aqua Base', 'Power Charm', false, 'none', false]
      },
      nerg_plains: {
        gouda: ['Gouda Cheese', 'best.weapon.tactical', 'Tribal Base,Aqua Base', 'Power Charm', false, 'none', false],
        boss: ['Gumbo Cheese', 'best.weapon.tactical', 'Tribal Base,Aqua Base', 'Power Charm', false, 'none', false]
      },
      elub_shore: {
        gouda: ['Gouda Cheese', 'Ancient Spear Gun,best.weapon.hydro', 'Tribal Base,Aqua Base', 'Power Charm', false, 'none', false],
        boss: ['Shell Cheese', 'Ancient Spear Gun,best.weapon.hydro', 'Tribal Base,Aqua Base', 'Power Charm', false, 'none', false]
      }
    }
  };
  const settings = isProd
    ? getStorageToObject(targetEnvironmentType, defaultSettings)
    : defaultSettings;
  // MH constants
  // const quest = isProd ? user.quests.QuestForewordFarm : QuestForewordFarm;
  // const trapState = quest.mice_state;
  const tribalIslesUtility = (() => {
    const utils = {
      // values
      URI: 'managers/ajax/environment/foreword_farm.php',
      /** Set immutable fields */
      setImmutable() {
        const immutableFields = ['URI'];
        for (let i = 0; i < immutableFields.length; i++) {
          const immutableField = immutableFields[i];
          Object.defineProperty(utils, immutableField, {
            // value: 'This cannot be changed',
            writable: false, // Makes the field immutable
            configurable: false, // Prevents deletion or re-configuration of the field
            enumerable: true // Makes the field visible during enumeration (e.g., in a for...in loop)
          });
        }
      },
      /** trapState, default: xxx */
      trapState: gouda,
      /** start running */
      run() {
        logging('start running tribalIsles');
        this.displayItemPromise();
      },
      displayItemPromise() {
        logging('start running displayItemPromise');
        return new Promise((resolve, reject) => {
          const latestEntryId = $('#journallatestentry').data('entry-id');
          const entry = $(`div[data-entry-id=${latestEntryId}]`);
          const journalText = entry.find('.journaltext').text();
          console.log(entry, journalText);
          const lootKey =
            environmentType === derr
              ? delicious
              : environmentType === nerg
              ? savoury
              : seashell;
          if (journalText.indexOf(lootKey) > -1) {
            getQuantityByUrlThen(items[lootKey], (qty) => {
              console.log(`Have ${qty} ${lootKey}`);
              localStorage.displayedMessage = `${lootKey}: ${qty}`;
            });
          }
          this.autoSettingsPromise();
        });
      },
      /** auto settings */
      autoSettingsPromise() {
        logging('start running autoSettingsPromise');
        return new Promise((resolve, reject) => {
          // auto settings for running if necessary and then invoke next promise
          const baitItemId = getBaitItemId();
          if (
            (baitItemId === 84 && environmentType === derr) ||
            (baitItemId === 99 && environmentType === nerg) ||
            (baitItemId === 112 && environmentType === elub)
          )
            this.trapState = boss;
          else this.trapState = gouda;
          this.handleTrapPromise();
        });
      },
      /** handle trap promise */
      handleTrapPromise() {
        logging('start running handleTrapPromise');
        return new Promise((resolve, reject) => {
          const trap = settings.traps[environmentType][this.trapState];
          if (trap) armAllTraps(...trap);
        });
      },
      /** location ajax */
      ajax(element, params, successCallback, errorCallback) {
        params['last_read_journal_entry_id'] = $('#journallatestentry').data(
          'entry-id'
        );
        params['uh'] = user.unique_hash;
        $.ajax({
          data: params,
          dataType: 'json',
          type: isProd ? 'post' : 'get',
          url: isProd
            ? MOUSEHUNT_BASE_URL + this.URI
            : 'https://elie2201.github.io/mh/jobs.txt',
          success: function (data) {
            eventRegistry.doEvent('ajax_response', data);
            if (successCallback) {
              successCallback(data);
            }
          },
          error: function (data) {
            // eslint-disable-next-line new-cap
            const popup = new jsDialog();
            popup.setTemplate('error');
            popup.addToken('{*title*}', 'An error has occurred.');
            popup.addToken(
              '{*content*}',
              'An error occurred. Please try again!'
            );
            popup.show();
            if (errorCallback) errorCallback(data);
          }
        });
      }
    };
    // initial utils
    function init() {
      // make fields immutable
      utils.setImmutable();
      // initialize values in utils if necessary
    }
    init();
    return utils;
  })();
  tribalIslesUtility.run();
}
