function moussuPicchu() {
  const isProd = false;
  const targetEnvironmentType = 'moussu_picchu';
  const enviromentType = MhUtils.getEnvironmentType();
  if (enviromentType !== targetEnvironmentType) return;
  logging('moussuPicchu_1.0.4.0.0');
  // localStorage settings
  // prettier-ignore
  const defaultSetup = {
    huntingPlan: 'auto',
    reservedRainyQty: 220,
    reservedWindyQty: 220,
    goalShadowvine: 0,
    goalArcanevine: 0,
    trap: {
      none: ['Gouda Cheese', 'Boiling Cauldron', "Alchemist's Cookbook", 'Airship Charm,Super Snowball Charm', false],
      wind: [
        ['Windy Cheese', 'Boiling Cauldron', "Alchemist's Cookbook", 'Airship Charm,Tarnished Charm,Unstable', false, 35],
        ['Windy Cheese', 'best.weapon.arcane', 'best.base.combo', 'Ancient Charm', false, 80],
        ['Windy Cheese', 'best.weapon.arcane', 'best.base.combo', 'Regal Charm,Super Regal Charm,Ancient Charm', false, 94],
        ['Windy Cheese', 'best.weapon.arcane', 'best.base.combo', 'Ultimate Luck Charm,Festive Ultimate Luck Charm', false, 101]
      ],
      wind_minluck: [
        ['Windy Cheese', 'Boiling Cauldron', "Alchemist's Cookbook", 'Airship Charm,Tarnished Charm,Unstable', false, 35],
        ['Windy Cheese', 'best.weapon.arcane', 'Seasonal Gift', 'Extreme Chrome Charm', false, 80],
        ['Windy Cheese', 'best.weapon.arcane', 'best.base.combo', 'Ultimate Luck Charm,Festive Ultimate Luck Charm', false, 94],
        ['Windy Cheese', 'best.weapon.arcane', 'best.base.combo', 'Ultimate Luck Charm,Festive Ultimate Luck Charm', false, 101]
      ],
      rain: [
        ['Rainy Cheese', 'best.weapon.shadow', "Seasonal Gift", 'Airship Charm,Tarnished Charm,Unstable', false, 35],
        ['Rainy Cheese', 'best.weapon.shadow', 'best.base.combo', 'Ancient Charm', false, 80],
        ['Rainy Cheese', 'best.weapon.shadow', 'best.base.combo', 'Regal Charm,Super Regal Charm,Ancient Charm', false, 94],
        ['Rainy Cheese', 'best.weapon.shadow', 'best.base.combo', 'Ultimate Luck Charm,Festive Ultimate Luck Charm', false, 101]
      ],
      rain_minluck: [
        ['Rainy Cheese', 'best.weapon.shadow', "Seasonal Gift", 'Airship Charm,Tarnished Charm,Unstable', false, 35],
        ['Rainy Cheese', 'best.weapon.shadow', 'Seasonal Gift', 'Extreme Chrome Charm', false, 80],
        ['Rainy Cheese', 'best.weapon.shadow', 'best.base.combo', 'Ultimate Luck Charm,Festive Ultimate Luck Charm', false, 94],
        ['Rainy Cheese', 'best.weapon.shadow', 'best.base.combo', 'Ultimate Luck Charm,Festive Ultimate Luck Charm', false, 101]
      ],
      storm: [
        ['Dragonvine Cheese', 'best.weapon.draconic', "Seasonal Gift", 'Airship Charm,Luck Charm', false, 35],
        ['Dragonvine Cheese', 'best.weapon.draconic', 'best.base.combo', 'Dragonbane Charm', false, 80],
        ['Dragonvine Cheese', 'best.weapon.draconic', 'best.base.combo', 'Dragonbane Charm', true, 94],
        ['Dragonvine Cheese', 'best.weapon.draconic', 'best.base.combo', 'Dragonbane Charm', true, 101]
      ]
    }
  };
  const setup = isProd
    ? MyUtils.getStorageToObject(targetEnvironmentType, defaultSetup)
    : defaultSetup;
  const focus = autoFocus(setup); // setup.huntingPlan;
  if (!focus) {
    setTimeout(() => {
      MhUtils.travel('rift_furoma');
    }, 6000);
    return;
  }
  if (focus == 'none') {
    armAllTraps(...setup.trap.none);
    toggleTorch(setup.trap.none[4]);
    return;
  }
  const quest = isProd ? user.quests.QuestMoussuPicchu : QuestMoussuPicchu;
  const elementKey = focus.split('_')[0];
  let percent = quest.elements[elementKey].percent;
  if (elementKey == 'storm')
    percent = Math.min(
      quest.elements.rain.percent,
      quest.elements.wind.percent
    );
  const traps = setup.trap[focus];
  for (let i = 0; i < traps.length; i++) {
    const trap = traps[i];
    if (percent < trap[5]) {
      console.log('Choosen trap:', trap);
      armAllTraps(...trap);
      toggleTorch(trap[4]);
      return;
    }
  }
  function autoFocus(settings) {
    if (settings.huntingPlan != 'auto') return settings.huntingPlan;
    const quest = user.quests.QuestMoussuPicchu;
    const items = quest.items;
    const fuelQty = items.tiki_fuel_stat_item.quantity;
    const rainyCheeseQty = MyUtils.parseQuantity(items.rain_cheese.quantity);
    const windyCheeseQty = MyUtils.parseQuantity(items.wind_cheese.quantity);
    const dragonCheeseQty = MyUtils.parseQuantity(
      items.dragonvine_cheese.quantity
    );
    const shadowvineQty = MyUtils.parseQuantity(
      items.shadowvine_crafting_item.quantity
    );
    const arcanevineQty = MyUtils.parseQuantity(
      items.arcanevine_crafting_item.quantity
    );
    const rainPercent = quest.elements.rain.percent;
    const windPercent = quest.elements.wind.percent;
    // const stormPercent = Math.min(rainPercent, windPercent);
    // prettier-ignore
    const stages = ['rain','wind','wind_minluck','rain_minluck','storm','rain_minluck','storm'];
    let stage = localStorage.moussuPicchuStage;
    if (!stage) {
      stage = 0;
      localStorage.moussuPicchuStage = stage;
      return stages[stage];
    }
    if (stage == -1) return 'none';
    if (stage == 0) {
      console.log(`rainy until ${settings.reservedRainyQty} left => wind`);
      // TODO cheeseQty/percent設法寫活
      if (
        rainyCheeseQty > settings.reservedRainyQty &&
        (!settings.goalShadowvine || shadowvineQty < settings.goalShadowvine)
      )
        return stages[stage];
      if (rainPercent == 100) return stages[stage];
      stage = 1;
      localStorage.moussuPicchuStage = stage;
      return stages[stage];
    }
    if (stage == 1) {
      console.log(
        `windy until ${settings.reservedWindyQty} left => wind_minluck`
      );
      if (
        windyCheeseQty > settings.reservedWindyQty &&
        (!settings.goalArcanevine || arcanevineQty < settings.goalShadowvine)
      )
        return stages[stage];
      if (windPercent == 100) {
        stage = 3;
        localStorage.moussuPicchuStage = stage;
        return stages[stage];
      }
      stage = 2;
      localStorage.moussuPicchuStage = stage;
      return stages[stage];
    }
    if (stage == 2) {
      console.log('windy until wind 100 but not run out of windy');
      if (windPercent < 100)
        if (windyCheeseQty > 1) return stages[stage];
        else return null;
      stage = 3;
      localStorage.moussuPicchuStage = stage;
      return stages[stage];
    }
    if (stage == 3) {
      console.log('rainy to rain 18 and wind 82 => storm');
      // (18,82) (21,79)
      if (rainPercent < 18) return stages[stage];
      stage = 4;
      localStorage.moussuPicchuStage = stage;
      return stages[stage];
    }
    if (stage == 4) {
      console.log('DVC to rain 36 and wind 100 => rain_minluck');
      // (36,100)(39,100)(36,98) (36,94)(39,97)(34,92=>37,95)
      if (rainPercent < 36) return stages[stage];
      stage = 5;
      localStorage.moussuPicchuStage = stage;
      return stages[stage];
    }
    if (stage == 5) {
      console.log('rainy to rain 69 and wind 67 => storm');
      // (69,67)(72,64) (69,70)(72,67) (69,65)(72,62) (69,61)(72,58) (69,67)(70,62)
      if (Math.abs(rainPercent - windPercent) < 3) {
        stage = 6;
        localStorage.moussuPicchuStage = stage;
        return stages[stage];
      }
      if (rainPercent < 69) return stages[stage];
      stage = 6;
      localStorage.moussuPicchuStage = stage;
      return stages[stage];
    }
    if (stage == 6) {
      console.log('keep hunting with dragonvine.');
      if (dragonCheeseQty > 1 && fuelQty > 1) return stages[stage];
      return null;
    }
    return null;
  }
  // toggleTorch(false);
  function toggleTorch(isEnabled) {
    const quest = user.quests.QuestMoussuPicchu;
    if (MyUtils.parseQuantity(quest.items.tiki_fuel_stat_item.quantity) === 0)
      return;
    const isFuelEnabled = quest.is_torch_active;
    if ((isFuelEnabled == true) != isEnabled)
      hg.views.HeadsUpDisplayMoussuPicchuView.toggleTorch();
  }
}
