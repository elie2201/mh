import { mh, MhUtils, MyUtils } from './my-mh-aid-dev';

const hg = {};
const user = {};

// #FuromaRiftAddon
mh.addon.FuromaRift = class FuromaRift extends mh.AbstractBase {
  static version = `mh.addon.${FuromaRift.name}_1.3.1.0.0`;
  static targetEnvironmentType = 'rift_furoma';
  static masterBaits = [2097, 2096, 2102];
  static beltHeirloom = 2083;
  static clawHeirloom = 2089;
  static fangHeirloom = 2090;
  static eibType = 'furoma_rift_energy_base';
  static aeibType = 'furoma_rift_energy_upgraded_base';
  static aeibId = 2120;
  static async buyAeib() {
    const { log } = this;
    if (MhUtils.getEnvironmentType() !== this.targetEnvironmentType) return;
    if (MhUtils.getBaseItemId() === this.aeibId) return;
    log.log('this.buyAeib');
    if (!this.masterBaits.includes(MhUtils.getBaitItemId())) return;
    // Chi Belt Heirloom
    let qty = await mh.UserInventory.getItemQuantityPromise(this.beltHeirloom);
    if (qty < 1) return;
    log.log('Chi Belt Heirloom done');
    // Chi Claw Heirloom
    qty = await mh.UserInventory.getItemQuantityPromise(this.clawHeirloom);
    if (qty < 1) return;
    log.log('Chi Claw Heirloom done');
    // Chi Fang Heirloom
    qty = await mh.UserInventory.getItemQuantityPromise(this.fangHeirloom);
    if (qty < 1) return;
    log.log('Chi Fang Heirloom done');
    await mh.Shop.completeTransactionPromise(true, this.eibType, 1, false);
    await mh.Shop.completeTransactionPromise(true, this.aeibType, 1, false);
    hg.utils.TrapControl.setBase(2120).go(() => {
      mh.User.travelPromise('rift_bristle_woods').catch(() => {});
    });
  }
  static async spendEnerchiCharm(
    diff = 11,
    enoughEcQuantity = 0,
    travelTo = 'rift_whisker_woods'
  ) {
    const { log } = this;
    if (MhUtils.getEnvironmentType() !== this.targetEnvironmentType) return;
    log.log(`run spendEnerchiCharm(${diff}, ${enoughEcQuantity}, ${travelTo})`);
    const quest = user.quests.QuestRiftFuroma;
    if (quest.view_state.includes('pagoda')) return;
    log.log('Outside pagoda');
    const items = quest.items;
    const ec = MyUtils.parseQuantity(items.rift_furoma_energy_trinket.quantity);
    if (ec < enoughEcQuantity) return;
    log.log('Enough Enerchi Charm');
    // 兩種不同的計算 max battery total的方式,結果應該是一樣的
    const max1 = Object.values(user.quests.QuestRiftFuroma.batteries).reduce(
      (max, b) =>
        b.status.startsWith('unlocked') && b.total > max ? b.total : max,
      0
    );
    const max2 = Math.max(
      0,
      ...Object.values(user.quests.QuestRiftFuroma.batteries)
        .filter((b) => b.status.startsWith('unlocked'))
        .map((b) => b.total)
    );
    log.log(max1, max2);
    const max = Math.max(max1, max2);
    const enerchi = MyUtils.parseQuantity(
      items?.combat_energy_stat_item?.quantity
    );
    log.log(enerchi, max);
    if (enerchi > max - diff) MhUtils.pauseByTraveling(travelTo);
  }
  static async spendEcWithEffEnerchi(
    effEnerchiWanted = 4100,
    enoughEc = 5,
    travelTo = 'rift_whisker_woods'
  ) {
    const { log } = this;
    if (MhUtils.getEnvironmentType() !== this.targetEnvironmentType) return;
    log.log(`${this.spendEcWithEffEnerchi.name}`, {
      effEnerchiWanted,
      enoughEc,
      travelTo
    });
    const quest = user.quests.QuestRiftFuroma;
    if (quest.view_state.includes('pagoda')) return;
    log.log('Outside pagoda');
    const items = quest.items;
    const ec = MyUtils.parseQuantity(items.rift_furoma_energy_trinket.quantity);
    if (ec < enoughEc) return;
    log.log(`Enough Enerchi Charm`);
    const enerchi = MyUtils.parseQuantity(
      items?.combat_energy_stat_item?.quantity
    );
    log.log(enerchi, ec);
    if (enerchi + ec * 5 > effEnerchiWanted) MhUtils.pauseByTraveling(travelTo);
  }
};
