import { mh, MyUtils, MhUtils } from './my-mh-aid-dev';

const user = {};
mh.addon = {};

// #addonFloatingIslands
mh.addon.FloatingIslands = class FloatingIslands {
  static version = `mh.addon.${FloatingIslands.name}_1.0.0.0.2`;
  constructor() {
    console.info(`${FloatingIslands.version} initialized`);
  }
  static async hideCyclone() {
    console.info(`run hideCyclone`);
    const atts = user.quests.QuestFloatingIslands.hunting_site_atts;
    const isLaunch = atts.island_type.includes('launch');
    const cyclone = MyUtils.parseQuantity(
      user?.enviroment_atts?.items?.sky_scrambler_stat_item?.quantity
    );
    const hasCyclone = cyclone > 0;
    if (isLaunch && !hasCyclone && atts.sky_wardens_caught == 4) {
      await mh.addon.Marketplace.cancelAndResellCyclone(60_000, false);
      return;
    }
    if (!hasCyclone) return;
    if (
      atts.island_type.includes('tactical') ||
      atts.island_mod_types.includes('sky_cheese') ||
      atts.island_mod_types.includes('sky_pirates')
    )
      await mh.addon.Marketplace.cancelAndResellCyclone(60_000);
  }
  static updateGPStats() {
    console.info(`run updateGPW`);
    const atts = user.quests.QuestFloatingIslands.hunting_site_atts;
    const isLaunch = atts.island_type.includes('launch');
    if (isLaunch && atts.sky_wardens_caught == 4) MhUtils.updateGPStats();
  }
};
if (user.quests.QuestFloatingIslands)
  (async () => {
    await mh.addon.FloatingIslands.hideCyclone();
    mh.addon.FloatingIslands.updateGPStats();
  })();
// End #addonFloatingIslands
