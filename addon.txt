function autoSkyMap(settings) {

  const items = env.items;
  const stormCell = items.airship_rocket_fuel_stat_item;
  const stormCellQty = parseQuantity(stormCell.quantity);
  const cyclone = items.sky_scrambler_stat_item;
  const cycloneQty = parseQuantity(cyclone.quantity);
  if (
    (!settings.isAutoLai || atts.sky_wardens_caught == 4) &&
    (!settings.isAutoHai || atts.sky_wardens_caught < 4) &&
    (!settings.isAutoSp || stormCellQty < 50)
  ) {
    console.log(
      'Not auto sky map. Note if all 4 wardens are caught, there is not LAI to auto'
    );
    return;
  }
  console.plog(`Cyclone quantity before sky map: ${cycloneQty}`);
  if (settings.isAutoSp && stormCellQty > 49) {
    hg.views.FloatingIslandsAdventureBoardView.showSkyPalace(); // TODO 測試時 comment掉
    const invl = setInterval(() => {
      const actives = $('div[class*="active"]', $('#overlayPopup'));
      // TODO 測試用 const actives = $('div[class*="active"]', $('#overlayPopup', $(html)));
      if (actives.length < 5) {
        return;
      }
      clearInterval(invl);

  } else {

      if (tiles.length != 16) {

      }
      clearInterval(invl);
      return treatSkyMap(settings);
    }, 1000);
  }
}
function treatSkyMap(settings) {
  const quest = user.quests.QuestFloatingIslands;
  const huntingSiteAtts = quest.hunting_site_atts;
  const isSkyMapLaunched = true;
  const tiles = $('div[data-type]', $('#overlayPopup'));
  const islands = {};
  for (let i = 0; i < 4; i++) {
    const powerType = AllPowerTypes[i];
    const j = i * 4;
    islands[powerType] = [
      tiles[j].getAttribute('data-type'),
      tiles[j + 1].getAttribute('data-type'),
      tiles[j + 2].getAttribute('data-type'),
      tiles[j + 3].getAttribute('data-type')
    ];
  }
  for (let i = 4; i < AllPowerTypes.length; i++) {
    const powerType = AllPowerTypes[i];
    const j = i % 4;
    islands[powerType] = [
      tiles[j + 12].getAttribute('data-type'),
      tiles[j + 8].getAttribute('data-type'),
      tiles[j + 4].getAttribute('data-type'),
      tiles[j].getAttribute('data-type')
    ];
  }
  console.log(islands);
  // 兩種處理方式:分 laiSkyMap跟 haiSkyMap或用 shrineKey區分 lai/hai.
  // lai跟 hai的 tiles只差在神殿
  if (huntingSiteAtts.sky_wardens_caught != 4 && settings.isAutoLai)
    return skyMap(islands, settings);
  if (huntingSiteAtts.sky_wardens_caught == 4 && settings.isAutoHai)
    return skyMap(islands, settings, true);
  return !isSkyMapLaunched;
}
function skyMap(islands, settings, isHai) {
  const isSkyMapLaunched = true; // return true to stop invoker(from reloading/trap arming in launch)
  const myPowerTypes = isHai
    ? settings.myHaiPowerTypes
    : settings.myLaiPowerTypes;
  const shrineKey = isHai ? 'paragon_cache' : 'shrine';
  const included = settings.includedHaiShrine;
  const choices = [];
  // 防呆:測試時曾經忘記放 shrineMaxIndex導致無限 cycle
  const shrineMaxIndex = settings.shrineMaxIndex ? settings.shrineMaxIndex : 2;
  for (let i = 0; i < shrineMaxIndex; i++) {
    for (let j = 0; j < myPowerTypes.length; j++) {
      const myPowerType = myPowerTypes[j];
      const modifier = islands[myPowerType][i];
      // if (modifier.indexOf('shrine') > -1) {
      if (modifier.indexOf(shrineKey) > -1) {
        const isFilterOk =
          !included ||
          included.length == 0 || // 防呆: included中 4個 shrine都沒有的話會無限 cycle
          included.indexOf(modifier) > -1;
        console.log(`isHaiOk? ${isFilterOk}`);
        if (!isHai || isFilterOk) choices.push(myPowerType);
      }
    }
  }
  if (choices.length > 0) {
    const choice = bestChoice(islands, choices, settings.wantedModifier);
    console.log('launchSkyMap', choice);
    if (choice) {
      launchSkyMap(choice); // TODO 測試時 comment掉
      return isSkyMapLaunched;
    }
  }
  // Use Cyclone or choose third-tile island?
  // If use cyclone: terminate running autoSkyMap > confirm cycled > call treatSkyMap in cycled again
  // TODO 不測試 cycled()時用這行 const isCycle = false;
  const isCycle = isHai ? settings.isCycleHai : settings.isCycleLai;

function cycled(settings) {
  const getTileTypes = () => {
    const tiles = $('div[data-type]', $('#overlayPopup'));
    const tileTypes = [];
    tiles.each((i, e) => {
      tileTypes.push(e.getAttribute('data-type'));
    });
    return tileTypes;
  };
  const sameArray = (a, a1) => {
    for (let i = 0; i < a.length; i++) {
      const element = a[i];
      if (!a1[i] || a1[i] !== element) return false;
    }
    return true;
  };
  const tileTypes = getTileTypes();
  const rerollClass = '.floatingIslandsAdventureBoardSkyMap-rerollButton';
  // document.querySelector(rerollClass).click();
  document.querySelector(rerollClass).dispatchEvent(new Event('click'));
  const key = setInterval(() => {
    const newTileTypes = getTileTypes();
    if (sameArray(newTileTypes, tileTypes)) {
      return;
    }
    clearInterval(key);
    treatSkyMap(settings);
  }, 1000);
}
