// MH & my-mh-aid functions
function getEnvironmentType() {
  const rtn = trimToEmpty(user.environment_type);
  // console.plog(`Current location environment_type: ${rtn}`);
  return rtn;
}
function trimToEmpty(str) {
  return trimToDefault(str);
}
function trimToDefault(str, deflt) {
  return str === 0 ? '0' : ((str || (deflt ? deflt : '')) + '').trim();
}
function logging(...any) {
  const debug = true;
  if (debug) {
    console.groupCollapsed(new Date(), ...any);
    console.log(Error().stack);
    console.groupEnd();
  }
}
function parseQuantity(qty) {
  const strQty = (qty || 0) + '';
  const rtn = parseInt(strQty.replaceAll(',', ''));
  return Number.isNaN(rtn) ? 0 : rtn;
}
function completeTransaction(
  isPurchase,
  itemType,
  itemQuantity,
  isKingsCartItem,
  success
) {
  if (itemQuantity < 1) {
    console.plog('Cannot buy 0 item');
    return;
  }
  const uniqueHash = user.unique_hash;
  $.ajax({
    data: {
      type: itemType,
      quantity: itemQuantity,
      buy: isPurchase ? 1 : 0,
      is_kings_cart_item: isKingsCartItem ? 1 : 0,
      uh: uniqueHash
    },
    dataType: 'json',
    type: 'post',
    url: MOUSEHUNT_BASE_URL + 'managers/ajax/purchases/itempurchase.php',
    success: function (data) {
      console.log(data.success, data);
      eventRegistry.doEvent('ajax_response', data);
      if (success) success();
    },
    error: function (data) {
      // const message = `completeTransaction for ${itemType} error`;
      // popupMessage(message);
      // localStorage.setItem(storingMessageKey, message);
      console.log(data);
    }
  });
}

function autoBuyCCDT() {
  const isProd = false;
  const targetEnvironmentType = 'rift_valour';
  const enviromentType = getEnvironmentType();
  if (enviromentType !== targetEnvironmentType) return;
  if (parseInt(user.weapon_item_id) !== 2342) return;
  logging('autoBuyCCDT_2.0.0.0.0');
  // MH constants
  const quest = isProd ? user.quests.QuestRiftValour : QuestRiftValour;
  const autoBuyCCDTUtility = (() => {
    const utils = {
      // values
      /** Set immutable fields */
      setImmutable() {},
      /** trapState, default: xxx */
      trapState: 'xxx',
      /** start running */
      run() {
        logging('start autoBuyCCDT');
        this.autoSettingsPromise();
      },
      /** auto settings */
      autoSettingsPromise() {
        logging('start autoSettingsPromise');
        return new Promise((resolve, reject) => {
          // auto settings for running if necessary and then invoke next promise
          if (quest.is_at_eclipse) return;
          const finalItemQty = parseQuantity(
            document.querySelector(
              'div[data-item-type="total_eclipse_resource_stat_item"]'
            ).innerText
          );
          if (parseInt(user.base_item_id) === 3080) {
            if (quest.power_up_data.long_stride.current_value < 9) return;
            if (finalItemQty < 2) return;
            console.log('After speed 9 with 2 final crafting items up, buy PB');
            const itemType = 'valour_rift_prestige_base';
            completeTransaction(true, itemType, 1, false, () => {
              hg.utils.TrapControl.armItem(itemType, 'base').go();
            });
            return;
          }
          if (parseInt(parseInt(user.weapon_item_id)) !== 2342) {
            console.log('Not arming TDT, ignore buying weapon');
            return;
          }
          if (finalItemQty < 20) {
            console.log('Not have enough final crafting item');
            return;
          }
          console.log('Have 20 final crafting items up');
          completeTransaction(
            true,
            'celestial_dissonance_weapon',
            1,
            false,
            () => {
              const itemType = 'celestial_dissonance_weapon';
              const confirmQuantity = 1;
              hg.utils.UserInventory.smashItem(
                itemType,
                confirmQuantity,
                () => {
                  let items = {
                    celestial_dissonance_trap_parts_crafting_item: 1
                  };
                  hg.utils.UserInventory.craftItem(
                    items,
                    confirmQuantity,
                    () => {
                      hg.utils.UserInventory.smashItem(
                        itemType,
                        confirmQuantity,
                        () => {
                          items = {
                            celestial_dissonance_trap_parts_crafting_item: 1,
                            chrome_celestial_dissonance_kit_crafting_item: 1
                          };
                          hg.utils.UserInventory.craftItem(
                            items,
                            confirmQuantity,
                            () => {
                              hg.utils.TrapControl.armItem(
                                'chrome_celestial_dissonance_weapon',
                                'weapon'
                              ).go();
                            }
                          );
                        }
                      );
                    }
                  );
                }
              );
            }
          );
        });
      }
    };
    // initial utils
    function init() {
      // make fields immutable
      utils.setImmutable();
      // initialize values in utils if necessary
    }
    init();
    return utils;
  })();
  autoBuyCCDTUtility.run();
}
