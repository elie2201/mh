if (user.environment_type == 'desert_city') muridaeMarket();
function muridaeMarket() {
  // const isProd = false;
  const targetEnvironmentType = 'desert_city';
  const enviromentType = getEnvironmentType();
  if (enviromentType !== targetEnvironmentType) return;
  logging('muridaeMarket_1.0.0.1.1');
  defaultBait = 'Gouda Cheese';
  const quest = user.quests.QuestDesertMarketRebuild;
  const items = quest.items;
  const papyrusQty = items.papyrus_crafting_item;
  const sort = 'sort';
  const name = 'name';
  const settings = JSON.parse(localStorage['locationSetup'])[enviromentType];
  const charmName = getCharmName();
  const muridaeMarketUtils = {
    run() {
      logging('start running muridaeMarket');
      this.initPromise();
    },
    initPromise() {
      logging('run initPromise');
      return new Promise((resolve, reject) => {
        this.checkArtisanPromise();
      });
    },
    checkArtisanPromise() {
      logging('run checkArtisanPromise');
      if (charmName.indexOf('Artisan') == 0 && papyrusQty > 24)
        settings[TrapType.trinket][name] = '2025 Charm';
      this.handleTrapPromise();
    },
    handleTrapPromise() {
      for (const key in TrapType) {
        if (Object.prototype.hasOwnProperty.call(TrapType, key)) {
          const trapType = TrapType[key];
          checkThenArm(
            settings[trapType][sort],
            trapType,
            settings[trapType][name]
          );
        }
      }
    }
  };
  muridaeMarketUtils.run();
}
