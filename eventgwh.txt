  function ajax(element, params, uri) {
    console.log('GWH ajax');
    return new Promise((resolve, reject) => {
      const callbackurl = 'https://www.mousehuntgame.com/';
      const lastReadJournalEntryId = parseInt(
        document
          .querySelector('div[data-entry-id]')
          .getAttribute('data-entry-id')
      );
      params['last_read_journal_entry_id'] = lastReadJournalEntryId;
      params['uh'] = user.unique_hash;
      $.ajax({
        data: params,
        dataType: 'json',
        type: 'post',
        url: callbackurl + uri, // 'managers/ajax/events/winter_hunt_region.php',
        success: function (data) {
          if (data.success) {
            resolve(data);
          } else {
            reject(data);
          }
        },
        error: function () {
          // eslint-disable-next-line new-cap
          const popup = new jsDialog();
          popup.setTemplate('error');
          popup.addToken('{*title*}', 'An error has occurred.');
          popup.addToken('{*content*}', 'An error occurred. Please try again!');
          popup.show();
          // target.removeClass('busy');
          // eslint-disable-next-line prefer-promise-reject-errors
          reject();
        }
      });
    });
  }

  function toggleCannons(booleanArray, from, to) {
    const enviromentType = user.environment_type || '';
    console.log('currently in', enviromentType);
    if (enviromentType != 'winter_hunt_fortress') return;
    console.log(booleanArray, from, to);
    // const begin = from || 0;
    // const end = to || 2;
    let begin = from;
    let end = to;
    if (begin != 0 && !begin) begin = 0;
    if (end != 0 && !end) end = 2;
    if (begin > end) return;
    console.log('need to try toggle cannons', begin, end);
    let boolArray = booleanArray;
    if (!boolArray) boolArray = [false, false, false];
    toggleCannon(begin, boolArray[begin]);
    setTimeout(() => {
      toggleCannons(boolArray, ++begin, end);
    }, Math.floor(400 + Math.random() * 300));
  }
  function toggleCannon(index, isEnable) {
    const enviromentType = user.environment_type;
    console.log('currently in', enviromentType);
    if (enviromentType != 'winter_hunt_fortress') return;
    console.log('right location, try toggling');
    const quest = user.quests.QuestIceFortress;

      quest.cannons.snow_cannon.is_enabled || false,
      quest.cannons.cinnamon_cannon.is_enabled || false,
      quest.cannons.charm_cannon.is_enabled || false
    ];
    if (cannonsState[index] == isEnable) return;
    console.log('need to try toggle cannon', index);
    const cannonTypes = ['snow', 'cinnamon', 'charm'];
    const cannonType = cannonTypes[index];
    const uri = 'managers/ajax/events/winter_hunt_ice_fortress.php';
    const params = { action: 'toggle_cannon', cannon_type: cannonType };
    ajax(null, params, uri);
  }

  function checkGolem(setup) {
    if (!setup.isAutoGolem) return;
    const quest = getQuest();
    console.log('Run checkGolem', quest);
    const golems = quest.golems;
    for (let i = 0; i < setup.maxGolemsUse; i++) {
      const golem = golems[i];
      if (golem.can_upgrade) {
        console.log('upgrade golem');
        upgradeGolem(golem.slot);
        golem.can_upgrade = false;
        setTimeout(() => {
          checkGolem(setup);
        }, Math.floor(setup.minGolemDelayTime + Math.random() * (setup.maxGolemDelayTime - setup.minGolemDelayTime)));
        return;
      }
    }
    for (let i = 0; i < setup.maxGolemsUse; i++) {
      const golem = golems[i];
      if (golem.can_claim) {
        console.log('claim golem');
        claimReward(golem.slot);
        golem.can_claim = false;
        setTimeout(() => {
          checkGolem(setup);
        }, Math.floor(setup.minGolemDelayTime + Math.random() * (setup.maxGolemDelayTime - setup.minGolemDelayTime)));
        return;
      }
    }
    for (let i = 0; i < setup.maxGolemsUse; i++) {
      const golem = golems[i];
      if (golem.can_build) {
        console.log('build golem');
        buildGolem(golem.slot, setup.golemSetup.defaultTarget[i]);
        golem.can_build = false;
        setTimeout(() => {
          checkGolem(setup);
        }, Math.floor(setup.minGolemDelayTime + Math.random() * (setup.maxGolemDelayTime - setup.minGolemDelayTime)));
        return;
      }
    }
    console.log('End checkGolem');
    /* const popup = new jsDialog();
    popup.addToken('{*title*}', 'End checkGolem.');
    popup.addToken('{*content*}', 'Check golem finished');
    popup.show(); */
  }

  function buildGolem(slot, environmentType) {
    const scarfQty = getQuest().items.golden_scarf_stat_item.quantity;
    let scarfingDestination = null;
    let scarfingDestinations = null;
    if (scarfQty > 0) {
      scarfingDestinations = localStorage.scarfingDestinations;
      if (scarfingDestinations) {
        scarfingDestinations = JSON.parse(scarfingDestinations);
        if (scarfingDestinations.length > 0 && scarfingDestinations[0][1] != '')
          if (
            scarfingDestinations[0][0] == -1 ||
            scarfingDestinations[0][0] == slot
          )
            scarfingDestination = scarfingDestinations.shift(0)[1];
      }
    }
    const isScarf = scarfingDestination ? true : false;
