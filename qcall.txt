  /**
   * 檢查是否能建造指定的 Cork。
   * @param {string} targetCork - 目標 Cork 名稱。
   * @returns {string|null} - 可建造的 Cork 名稱；若材料不足或 bark 不足則回傳 null。
   */
  function getGeyserBuildCork(targetCork) {
    console.log(`Can build ${targetCork}?`);
    const quest = user.quests.QuestQuesoGeyser;
    const startIndex = Math.max(Corks.indexOf(targetCork), 0);
    for (let i = startIndex; i < Corks.length; i++) {
      const corkType = Corks[i];
      const cork = quest.corks[corkType];
      const requirements = cork.build_requirements;
      const myItems = quest.items;
      const { isBarkEnough, isEnough } = checkRequirements(
        requirements,
        myItems
      );
      console.log(
        `${isBarkEnough && isEnough ? 'Can' : 'Cannot'} build ${corkType}`
      );
      // 如果只是 cork_bark_crafting_item數量不足,直接回傳 null,打斷 autoBuild
      // 不然會在 low rank cork一直把 cork bark消耗掉
      if (isEnough) {
        return isBarkEnough ? corkType : null;
      }
    }
    return null;
  }
  /**
   * 檢查建造需求是否滿足。
   * @param {Object} requirements - 建造需求物件，每個 key 對應一個需求。
   * @param {Object} myItems - 玩家持有的物品清單。
   * @returns {{isBarkEnough: boolean, isEnough: boolean}} - cork bark 是否足夠以及，其他材料是否足夠。
   */
  function checkRequirements(requirements, myItems) {
    let isEnough = true;
    let isBarkEnough = true;
    for (const key in requirements) {
      if (!Object.hasOwn(requirements, key)) continue;
      // 使用 alias，避免 ESLint camelCase 警告
      const { item_type: itemType, quantity } = requirements[key];
      const myQty = MyUtils.parseQuantity(myItems[itemType].quantity);
      if (itemType === 'cork_bark_crafting_item') {
        if (myQty < quantity) isBarkEnough = false;
      } else {
        // 保留原始邏輯：逐一累積判斷
        isEnough &&= myQty >= quantity;
      }
    }
    return { isBarkEnough, isEnough };
  }

  function getQuesoBaitType(baitName) {
    for (const key in QuesoBaitTypeNameMap) {
      if (Object.hasOwn(QuesoBaitTypeNameMap, key)) {
        const theName = QuesoBaitTypeNameMap[key];
        if (theName == baitName) return key;
      }
    }
    return null;
  }

  function isEnoughQuesoCheese(baitType, quest) {
    logging(`Is enough ${baitType}?`);
    if (
      baitType == QuesoBaitTypeNameMap.gouda_cheese ||
      baitType == QuesoBaitTypeNameMap.super_brie_cheese
    )
      return true;
    const myItems = quest.items;
    const baitQty = parseQuantity(myItems[baitType].quantity);
    const isEnough = baitQty >= EnoughCheeseQuantity;
    logging(`${isEnough ? '' : 'not'} enough ${baitType}`);
    return isEnough;
  }
  // End isEnoughQuesoCheese
  //* getLowerAvailableQuesoCheese('medium_queso_cheese');
  function getLowerAvailableQuesoCheese(quesoBaitType, quest) {
    logging(`Get lower available cheese of ${quesoBaitType}`);
    const getQuesoPreviousBaitType = (baitType) => {
      let armingOrder = QuesoBaitsOrder.indexOf(baitType);
      if (armingOrder < 0) return null;
      armingOrder -= 1;
      if (armingOrder < 0) {
        return null;
      }
      return QuesoBaitsOrder[armingOrder];
    };
    const myItems = quest.items;
    let baitType = quesoBaitType;
    let baitQty = parseQuantity(myItems[baitType].quantity);
    if (baitQty >= EnoughCheeseQuantity) {
      return baitType;
    }
    while (baitQty < EnoughCheeseQuantity) {
      baitType = getQuesoPreviousBaitType(baitType);
      if (!baitType) {
        travel(QuesoCanyonEnvironmentTypes.river);
        return baitType;
      }
      baitQty = parseQuantity(myItems[baitType].quantity);
    }
    logging(`Get ${baitType}`);
    return baitType;
  }

    if (!quest) quest = user.quests.QuestQuesoGeyser;

    const params = { action: 'toggle_wild_tonic' };
    if (user.quests.QuestQuesoCanyon) {
      const isFuelEnabled = quest.is_wild_tonic_active || false;
      if ((String(isFuelEnabled) === 'true') != isEnabled)
        hg.views.HeadsUpDisplayQuesoCanyonView.ajax('<a/>', params, callback);
    } else {
      const isFuelEnabled = quest.is_wild_tonic_enabled || false;
      if ((String(isFuelEnabled) === 'true') != isEnabled)
        hg.views.HeadsUpDisplayQuesoGeyserView.ajax('<a/>', params, callback);
    }
  }

      if (itemQty >= needQty) continue;
      armingBait = QuesoBaitsOrder[i];
      break;
    }
    // if not enough cheese, try lower cheese till bland_queso_cheese
    while (armingBait && !isEnoughQuesoCheese(armingBait, quest)) {
      armingBait = getLowerAvailableQuesoCheese(armingBait, quest);
    }
    if (!armingBait) {
      travel(QuesoCanyonEnvironmentTypes.river);
      return;
    }
    const trap = settings.traps[armingBait];
    armAllTraps(...trap);
    const isFuelEnabled = trap[4];

function parseQuantity(qty) {
  const strQty = (qty || 0) + '';
  const rtn = Number.parseInt(strQty.replaceAll(',', ''));
  return Number.isNaN(rtn) ? 0 : rtn;
