function getStorageToObject(keyName, objDefault) {
  let obj = getStorage(keyName);
  let bCheckNewProp = true;
  if (isNullOrUndefined(obj)) {
    obj = JSON.stringify(objDefault);
    setStorage(keyName, obj);
    bCheckNewProp = false;
  }
  obj = JSON.parse(obj);
  if (bCheckNewProp) {
    if (assignMissingDefault(obj, objDefault)) {
      setStorage(keyName, JSON.stringify(obj));
    }
  }
  return obj;
  function getStorage(name) {
    // check if the web browser support HTML5 storage
    if ('localStorage' in window && !isNullOrUndefined(window.localStorage)) {
      return window.localStorage.getItem(name);
    }
  }
  function setStorage(name, value) {
    // check if the web browser support HTML5 storage
    if ('localStorage' in window && !isNullOrUndefined(window.localStorage)) {
      window.localStorage.setItem(name, value);
    }
  }
  function isNullOrUndefined(obj) {
    return (
      obj === null || obj === undefined || obj === 'null' || obj === 'undefined'
    );
  }
  function assignMissingDefault(target, source) {
    let isResave = false;
    for (const key in source) {
      if (
        Object.prototype.hasOwnProperty.call(source, key) &&
        !Object.prototype.hasOwnProperty.call(target, key)
      ) {
        const element = source[key];
        target[key] = element;
        isResave = true;
      }
    }
    return isResave;
  }
}

function parseQuantity(qty) {

/**
 * Query item quantity by http get and then execute callback
 * @param {string} itemType
 * @param {Function} callback
 * @param {Boolean} isAlwaysCallback true if run callback even when qty is NaN or 0
 */
function getQuantityByUrlThen(itemType, callback, isAlwaysCallback) {
  const url = MOUSEHUNT_BASE_URL + 'item.php?item_type=' + itemType;
  $.get(url, (data) => {
    const key = 'You Own: ';
    const text = data.split(key)[1];
    const key2 = '</div>';
    const endIndex = text.indexOf(key2);
    const qtyText = text.substring(0, endIndex).trim();
    logging(qtyText);
    const qty = parseQuantity(qtyText.split('/')[0]);
    if (!isAlwaysCallback && (Number.isNaN(qty) || qty < 1)) return;
    if (callback) callback(qty);
  });
}

      /** boiling cauldron promise */
      boilingCauldronPromise(minBrewingQty) {
        logging(`start boilingCauldronPromise(${minBrewingQty})`);
        return new Promise((resolve, reject) => {
          const cauldronUsage = [
            [0, 0, 0],
            [0, 0, 0]
          ];
          for (let i = 0; i < cauldrons.length; i++) {
            const cauldron = cauldrons[i];
            cauldronUsage[i][0] = cauldron.is_brewing ? cauldron.brew_time : 0;
            for (let j = 0; j < cauldron.queue.length; j++) {
              const queue = cauldron.queue[j];
              if (queue.type) cauldronUsage[i][j + 1] = 15;
            }
          }
          console.log('cauldronUsage:', cauldronUsage);
          const recipeTypes = [
            'cauldron_tier_4_recipe',
            'cauldron_tier_3_recipe',
            'cauldron_tier_2_recipe',
            'cauldron_tier_1_recipe',
            'halloween_extract_cauldron_recipe'
          ];
          const itemTypes = [
            'cauldron_tier_4_cheese',
            'cauldron_tier_3_cheese',
            'cauldron_tier_2_cheese',
            'cauldron_tier_1_cheese',
            'halloween_extract_stat_item'
          ];
          // 完成 reward track前,先全力 brew EE到完成 reward track.
          // let inEventMap = false;
          // const maps = user.quests.QuestRelicHunter.maps;
          // for (let i = 0; i < maps.length; i++) {
          //   const map = maps[i];
          //   if (map.map_class == MapClasses.event) inEventMap = true;
          // }
          // if (totalEE > 600 && isTrackingReward && !inEventMap) {
          if (isTrackingReward) {
            ingredients.unshift(ingredients.pop());
            recipeTypes.unshift(recipeTypes.pop());
            itemTypes.unshift(itemTypes.pop());
          }
          console.log('ingredients:', ingredients);
          if (
            cauldronUsage[0][0] + cauldronUsage[0][1] > minBrewingQty &&
            cauldronUsage[1][0] + cauldronUsage[1][1] > minBrewingQty
          )
            return;
          console.log(
            `Have cauldrons progress less than ${minBrewingQty} , fill it up!!`
          );
          for (let i = 0; i < ingredients.length; i++) {
            const ingredient = ingredients[i];
            if (ingredient < 15) continue;
            for (let j = 0; j < cauldronUsage.length; j++) {
              const sumFirst2Slot = cauldronUsage[j][0] + cauldronUsage[j][1];
              if (sumFirst2Slot > minBrewingQty) continue;
              // brew
              const s = `<a data-cauldron-index="${j}" data-recipe-type="${recipeTypes[i]}" data-item-type="${itemTypes[i]}"></a>`;
              // const s = '<a data-cauldron-index="' + k + '" data-recipe-type="' + recipeTypes[i] + '" data-item-type="' + itemTypes[i] + '"></a>';
              console.log(
                `brew ${itemTypes[i]} in cauldron ${j} with sum of first 2 slots ${sumFirst2Slot}`,
                s
              );
              hg.views.HeadsUpDisplayHalloweenBoilingCauldronView.brewRecipe(
                $(s),
                () => {
                  cauldronUsage[j][1] = 15;
                  setTimeout(() => {
                    this.boilingCauldronPromise(minBrewingQty);
                    // reloadWithMessage('After add brewing, reloading...');
                  }, 1000);
                }
              );
              // 問題還是有點多 setTimeout(() => { boilingCauldron(minBrewingQty); }, 10000);
              return;
            }
          }
        });
      },
      /** handle trap promise */
      handleTrapPromise() {
        logging('start handleTrapPromise.');
        return new Promise((resolve, reject) => {
          const traps = isTrackingReward ? settings.trap : settings.trapBoon;
          if (quest.is_shutdown || quest.is_final_shutdown) {
            // traps[4][0] = 'Brie Cheese'; // shutdown phase用 Brie就夠了, brew只看 hunts
            armAllTraps(...traps[4]); // shoppe cheese setup
            this.toggleIncense(traps[4][4]);
            return;
          }
          const minBaitQuantity = isTrackingReward
            ? settings.minBaitQuantityTracking
            : settings.minBaitQuantity;
          const ingredients = [
            parseQuantity(
              quest.items.cauldron_tier_4_ingredient_stat_item
                .quantity_unformatted
            ),
            parseQuantity(
              quest.items.cauldron_tier_3_ingredient_stat_item
                .quantity_unformatted
            ),
            parseQuantity(
              quest.items.cauldron_tier_2_ingredient_stat_item
                .quantity_unformatted
            ),
            parseQuantity(
              quest.items.cauldron_tier_1_ingredient_stat_item
                .quantity_unformatted
            ),
            parseQuantity(
              quest.items.cauldron_potion_ingredient_stat_item
                .quantity_unformatted
            )
          ];
          if (settings.isLootEnding) {
            // 接近停止 loot時,把 tier-4 tier-3材料歸零就好,可以繼續收集 Bonefort
            // 也就是只用 Gouda/Monterey hunting. Bonefort材料有剩也不是太浪費
            if (ingredients[0] % 15 > 1) logging('Normally hunt.');
            else if (ingredients[1] % 15 > 2) minBaitQuantity[1] = 1000;
            else {
              minBaitQuantity[1] = 1000;
              minBaitQuantity[2] = 1000;
            }
          }
          logging('minBaitQuantity:', minBaitQuantity);
          const cheeseStats = [
            parseQuantity(
              quest.items.cauldron_tier_4_cheese.quantity_unformatted
            ),
            parseQuantity(
              quest.items.cauldron_tier_3_cheese.quantity_unformatted
            ),
            parseQuantity(
              quest.items.cauldron_tier_2_cheese.quantity_unformatted
            ),
            parseQuantity(
              quest.items.cauldron_tier_1_cheese.quantity_unformatted
            ),
            1000
          ];
          let priorityBegin = settings.priorityBegin;
          if (this.mapMouseRemaining() < settings.mapEndingQuantity)
            priorityBegin = Math.max(
              settings.priorityBegin,
              settings.mapEndingPriorityBegin
            );
          // logging('Auto choose cheese');
          for (let i = priorityBegin; i < cheeseStats.length; i++) {
            const baitQuantity = cheeseStats[i];
            if (baitQuantity > minBaitQuantity[i]) {
              logging('found best bait with enough quantity, bait index:', i);
              const trap = traps[i];
              logging(trap);
              if (trap) {
                armAllTraps(...trap);
                this.toggleIncense(
                  isTrackingReward || settings.isIncenseTrackComplete
                    ? settings.trap[i][4]
                    : false
                );
              }
              break;
            }
          }
        });
      },
      mapMouseRemaining() {
        const maps = user.quests.QuestRelicHunter.maps;
        for (let i = 0; i < maps.length; i++) {
          const map = maps[i];
          if (map.map_class == 'event') {
            logging(
              'Remaining map mouse quantity:',
              map.num_total - map.num_found
            );
            return map.num_total - map.num_found;
          }
        }
        return 0;
      },
