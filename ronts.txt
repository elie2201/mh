function travelingShoppe() {
  console.log(`travelingShoppe_0.1.4.2.7`);
  //* const quest = user.quests.QuestRonza;
  const goudaType = 'gouda_cheese';
  const rockforthType = 'rockforth_cheese';
  // prettier-ignore
  const defaultSettings = {
    isShutdown: false,
    gotdDates: [],
    farmingRockforthTrap: goudaType,
    startRockforthQty: 39,
    endRockforthQty: 13,
    traps: {
      gouda_cheese: ['Gouda Cheese', 'best.weapon.shadow', 'Seasonal Gift', 'Luck Charm,Super Cactus Charm'],
      super_brie_cheese: ['SUPER', 'best.weapon.physical', 'best.base.combo', 'Luck Charm,Super Cactus Charm'],
      rockforth_cheese: ['Rockforth Cheese', 'Boiling Cauldron', "Alchemist's Cookbook", 'Cactus Charm,Luck Charm,Super Cactus Charm'],
      gotd: ['', '', 'Seasonal Gift', ''],
      shutdown: ['Gouda Cheese', 'best.weapon.shadow', 'Seasonal Gift', 'None']
    }
  };
  const storageKey = 'ronzas_traveling_shoppe';
  const settings = MyUtils.getStorageToObject(storageKey, defaultSettings);
  if (
    Date.now() >
    new Date(user.quests.QuestRonza.timer_bar_end_date).setMinutes(10)
  )
    settings.isShutdown = true;
  if (settings.isShutdown) {
    armAllTraps(...settings.traps.shutdown);
    return;
  }
  const today = new Date();
  const todate = today.toISOString().split('T')[0];
  const isGotd = settings.gotdDates.includes(todate);
  const baitItemId = user.bait_item_id;
  const rockforthItemId = 109;
  if (baitItemId !== rockforthItemId) {
    getQuantityByUrlThen(rockforthType, (qty) => {
      const trap =
        qty > settings.startRockforthQty
          ? settings.traps.rockforth_cheese
          : settings.traps[settings.farmingRockforthTrap];
      if (isGotd) handleGotd(trap);
      armAllTraps(...trap);
    });
    return;
  }
  let currentTrapKey;
  const baitQuantity = MyUtils.parseQuantity(user.bait_quantity);
  if (baitQuantity < settings.endRockforthQty)
    currentTrapKey = settings.farmingRockforthTrap;
  else currentTrapKey = rockforthType;
  // check using localStorage.currentTrapKey if not empty and in settings
  const trapKeys = Object.keys(settings.traps);
  const checkUsingLocalTrapKey = (currentKey, effectiveKeys) => {
    let localTrapKey = MyUtils.trimToEmpty(localStorage.currentTrapKey);
    if (!effectiveKeys.includes(localTrapKey)) {
      localTrapKey = '';
      localStorage.currentTrapKey = localTrapKey;
    }
    if (localTrapKey !== '') return localTrapKey;
    return currentKey;
  };
  currentTrapKey = checkUsingLocalTrapKey(currentTrapKey, trapKeys);
  const trap = settings.traps[currentTrapKey];
  if (isGotd) handleGotd(trap);
  armAllTraps(...trap);
  function handleGotd(trap) {
    const gotdTrap = settings.traps.gotd;
    for (let i = 0; i < 4; i++) {
      const gotd = gotdTrap[i];
      trap[i] = gotd === '' ? trap[i] : gotd;
    }
  }
}
