mh.dao.ForewordFarm = class ForewordFarm extends mh.dao.Repository {
  static version = `mh.dao.${ForewordFarm.name}_0.0.0.0.1`;
  static uri = 'managers/ajax/environment/foreword_farm.php';
  static ImmutableFields = [...super.ImmutableFields, 'uri'];
  static {
    super.basicStaticBlock();
  }
  /**
   * 開始種植
   * @param {HTMLElement|string} element - DOM 元素或字串
   * @returns {Promise<void>}
   */
  static async startGrowing(slot, plantType, element) {
    const { log, version } = this;
    log.info(`${version}.${this.startGrowing.name}`, {
      slot,
      plantType,
      element
    });
    const params = {
      action: 'start_growing',
      slot: slot,
      plant_type: plantType
    };
    const request = { key: this.uri, element: element, payload: params };
    const data = await mh.dao.AjaxProvider.send(request);
    log.debug('ajax_response', data);
    try {
      activejsDialog.hide();
    } catch (error) {
      log.debug(error);
    }
    return data;
  }
  /**
   * 領取收穫
   * @param {HTMLElement|string} element - DOM 元素或字串
   * @returns {Promise<void>}
   */
  static async claimHarvest(element) {
    const { log, version } = this;
    log.info(`${version}.${this.claimHarvest.name}`, { element });
    const params = { action: 'claim_harvest' };
    const request = { key: this.uri, element: element, payload: params };
    const data = await mh.dao.AjaxProvider.send(request);
    log.debug('ajax_response', data);
    return data;
  }
  /**
   * 切換 Condensed Creativity (CC)
   * @param {HTMLElement|string} element - DOM 元素或字串
   * @returns {Promise<object>}
   */
  static async toggleFuel(element) {
    const { log, version } = this;
    log.info(`${version}.${this.toggleFuel.name}`, { element });
    const uri = 'managers/ajax/environment/folklore_forest_region.php';
    const params = { action: 'toggle_fuel' };
    const request = { key: uri, element: element, payload: params };
    const data = await mh.dao.AjaxProvider.send(request);
    log.debug('ajax_response', data);
    return data;

};

mh.core.ForewordFarm = class ForewordFarm extends (
  mh.core.ControllerFactory.Controller
) {
  static version = `mh.core.${ForewordFarm.name}_0.0.0.0.1`;
  // --- target location ---
  static targetEnvironmentType = 'foreword_farm';
  // my own hard-coded constants
  static plantRowTypes = [
    ['ordinary_farm_plant', 'ordinary_magic_farm_plant'],
    ['legendary_farm_plant', 'legendary_magic_farm_plant'],
    ['twisted_legendary_farm_plant', 'twisted_legendary_magic_farm_plant']
  ];
  static ImmutableFields = [
    ...super.ImmutableFields,
    'targetEnvironmentType',
    'plantRowTypes'
  ];
  static {
    super.basicStaticBlock();
  }
  static async toggleFuel(isToEnable, element) {
    const { log, version } = this;
    log.info(`${version}.${this.toggleFuel.name}`, { isToEnable, element });
    const quest =
      user.quests.QuestForewordFarm ||
      user.quests.QuestProloguePond ||
      user.quests.QuestTableOfContents;
    const fuelQuantity = MyUtils.parseQuantity(
      quest.items[mh.data.FoFoConfigConstants.ccType].quantity
    );
    if (fuelQuantity < 1) return;
    const isFuelEnabled = quest.is_fuel_enabled || false;
    if ((isFuelEnabled === true) != isToEnable)
      await mh.cmd.ForewordFarm.getInstance().toggleFuel(element);
  }
  async init() {
    const { log, version } = this.constructor;
    log.info(`${version}.${this.init.name}`);
    this.environmentType = MhUtils.getEnvironmentType();
    this.strategy = mh.cmd.ForewordFarm.getInstance();
    // --- settings ---
    this.data = mh.data.ForewordFarm;
    this.config = mh.cmd.ConfigFactory.getInstanceReloaded(this.data).config;
    // --- quest ---
    /**
     * 當前任務資訊
     * @type {object}
     */
    this.quest = user.quests.QuestForewordFarm;
    this.c = this.data.C;
    // --- items ---
    /**
     * 玩家物品資訊
     * @type {object}
     */
    this.items = {
      gold: user.gold,
      cc: {
        type: this.c.ccType,
        qty: MyUtils.parseQuantity(this.quest.items[this.c.ccType].quantity)
      },
      ddc: {
        type: this.c.ddcType,
        qty: MyUtils.parseQuantity(this.quest.items[this.c.ddcType].quantity)
      },
      papyrus: {
        type: this.c.papyrusType,
        qty: MyUtils.parseQuantity(
          this.quest.items[this.c.papyrusType].quantity
        )
      },
      ink: {
        type: this.c.inkType,
        qty: MyUtils.parseQuantity(this.quest.items[this.c.inkType].quantity)
      },
      mulch: {
        type: this.c.mulchType,
        qty: MyUtils.parseQuantity(
          this.quest.items['mythical_mulch_stat_item'].quantity
        )
      },
      seed: {
        type: this.c.seedType,
        qty: MyUtils.parseQuantity(
          this.quest.items['papyrus_seed_stat_item'].quantity
        )
      },
      me: {
        type: this.c.meType,
        qty: MyUtils.parseQuantity(this.quest.items[this.c.meType].quantity)
      }
    };
    /** @type {number} 當前陷阱狀態 (預設 no plants) */
    this.trapState = this.quest.mice_state;
    this.plots = this.quest.plots;
    /** 各農田是否可種植 */
    this.plotsPlantable = [];
    /** 是否進行種植 */
    this.isPlantable = false;
    /** 各農田的種植進度(剩餘 hunts) */
    this.plantState = [];
    for (const plot of this.plots) {
      this.isPlantable ||= plot.can_plant_anything;
      this.plotsPlantable.push(plot.can_plant_anything);
      this.plantState.push(MyUtils.parseQuantity(plot.num_catches_remaining));
    }
    /** 農田是否對齊 */
    this.isPlotsAligned =
      Math.max(...this.plantState) === Math.min(...this.plantState);
    if (this.config.isPlant3AtOnce && !this.isPlotsAligned) {
      this.isPlantable = false;
      this.plotsPlantable = [false, false, false];
    }
    log.debug(
      'Plots plantable State: ',
      `isPlantable: ${this.isPlantable}`,
      'plotsPlantable: ',
      this.plotsPlantable,
      `isPlotsAligned: ${this.isPlotsAligned}`,
      'plantState: ',
      this.plantState
    );
    this.check = new mh.chk.Check(
      false,
      async () => user.quests.QuestForewordFarm.items
    ).loadPlainText(this.config.checkers);
    return this;
  }
  async getItems() {
    const { log, version } = this.constructor;
    log.info(`${version}.${this.getItems.name}`);
    return this.items;
  }
  async run() {
    if (this.environmentType !== ForewordFarm.targetEnvironmentType) return;
    const { log, version } = this.constructor;
    log.info(`${version}.${this.run.name}`);
    log.debug('1. 檢查領取收穫');
    if (this.quest.harvest_bin.can_claim && this.config.isAutoClaim) {
      log.debug('before claiming harvest');
      if (await this.check.any(mh.chk.Check.CheckPoints.BeforeOpen)) return;
      await this.strategy.claimHarvest();
      log.debug('after claiming harvest');
      if (await this.check.any(mh.chk.Check.CheckPoints.AfterOpen)) return;
      return (await this.init()).run();
    }
    log.debug('2. 自動設定');
    await this.autoConfig();
    log.debug('3. 檢查自動種植');
    log.debug(
      `Need to plant? isAutoPlant: ${this.config.isAutoPlant}, isPlantable: ${this.isPlantable}`,
      this.plotsPlantable
    );
    if (this.config.isAutoPlant && this.isPlantable) {
      const plantType =
        this.constructor.plantRowTypes[this.config.plantTypeRow][
          this.config.isPlantWithMe ? 1 : 0
        ];
      await this.startGrowing(plantType);
      return (await this.init()).run();
    }
    log.debug('4. 陷阱處理');
    await this.handleTrap();
  }
  async startGrowing(plantType, plotIndex = 0) {
    const { log, version } = this.constructor;
    log.info(`${version}.${this.run.name}`, { plantType, plotIndex });
    const plotPlantable = this.plotsPlantable[plotIndex];
    log.debug(`Is plot(${plotIndex}) plantable? ${plotPlantable}`);
    if (!plotPlantable) {
      await this.startGrowing(plotIndex + 1);
      return;
    }
    if (plotIndex === 0) log.debug(`Started growing`);
    log.debug(`Started growing plot ${plotIndex}`);
    const isOk = await this.strategy.startGrowing(plotIndex, plantType);
    log.debug(`plant plot(${plotIndex}) ${isOk ? 'success' : 'failed'}`);
    if (plotIndex < this.plotsPlantable.length - 1) {
      await this.startGrowing(plantType, plotIndex + 1);
      log.debug(`Finished growing plot ${plotIndex}`);
      return;
    }
    log.debug(`Finished growing`);
  }
  /**
   * 自動調整 config
   */
  async autoConfig() {
    const { log, version } = this.constructor;
    log.info(`${version}.${this.autoConfig.name}`);
    if (this.items.seed.qty < 30) this.config.isAutoPlant = false;
    if (this.config.isAutoPlantType)
      if (this.items.mulch.qty < 60) this.config.plantTypeRow = 0;
      else this.config.plantTypeRow = 1;
  }
  /**
   * 陷阱處理
   */
  async handleTrap() {
    const { log, version } = this.constructor;
    log.info(`${version}.${this.handleTrap.name}`);
    let trap = this.config.traps[this.trapState];
    if (!trap) trap = this.config.traps['no_plants'];
    await this.constructor.toggleFuel(trap[4]);
    // 執行蠟燭與雙蛋效果
    litCandle(trap[5]);
    doubleEgg(trap[6]);
    // 延遲執行陷阱裝備
    setTimeout(() => {
      armAllTraps(...trap);
    }, 1500);
  }
};
