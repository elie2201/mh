  run() {
    console.log(this.nows);
    for (let i = 0; i < this.nows.length; i++) {
      if (!this.nows[i + 1]) {
        console.log('At end, stop running');
        break;
      }
      const now = this.nows[i];
      if (now !== MhUtils.getEnvironmentType()) {
        console.log(`Not in ${now}, skip`);
        continue;
      }
      const loot = this.loots[i];
      const next = this.nows[i + 1];
      const action = this.actions[i];
      if (hasLootByItemName(loot)) MhUtils.travel(next, 2000, action);
      break;
    }
  }
