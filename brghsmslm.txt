function getStorageToObject(keyName, objDefault) {
  let obj = getStorage(keyName);
  let bCheckNewProp = true;
  if (isNullOrUndefined(obj)) {
    obj = JSON.stringify(objDefault);
    setStorage(keyName, obj);
    bCheckNewProp = false;
  }
  obj = JSON.parse(obj);
  if (bCheckNewProp) {
    if (assignMissingDefault(obj, objDefault)) {
      setStorage(keyName, JSON.stringify(obj));
    }
  }
  return obj;
  function getStorage(name) {
    // check if the web browser support HTML5 storage
    if ('localStorage' in window && !isNullOrUndefined(window.localStorage)) {
      return window.localStorage.getItem(name);
    }
  }
  function setStorage(name, value) {
    // check if the web browser support HTML5 storage
    if ('localStorage' in window && !isNullOrUndefined(window.localStorage)) {
      window.localStorage.setItem(name, value);
    }
  }
  function isNullOrUndefined(obj) {
    return (
      obj === null || obj === undefined || obj === 'null' || obj === 'undefined'
    );
  }
  function assignMissingDefault(target, source) {
    let isResave = false;
    for (const key in source) {
      if (
        Object.prototype.hasOwnProperty.call(source, key) &&
        !Object.prototype.hasOwnProperty.call(target, key)
      ) {
        const element = source[key];
        target[key] = element;
        isResave = true;
      }
    }
    return isResave;
  }
}

function mousoleum() {
  logging('mousoleum_0.0.0.0.1');
  const isProd = false;
  const targetEnvironmentType = 'mousoleum';
  const enviromentType = getEnvironmentType();
  if (enviromentType !== targetEnvironmentType) return;
  // trap: [<bait>, <weapon>, <base>, <trinket>, <championFireEnabled>, litCandle]
  // prettier-ignore
  const defaultSetting = {
    isAutoBuild: false,
    isAutoRepair: true,
    trap: {
      radioactive_blue_cheese: ['Radioactive Blue Cheese', 'Boiling Cauldron', "event,Alchemist's Cookbook Base", 'None', false, 'none'],
      radioactive_blue_cheese_wall: ['Radioactive Blue Cheese', 'Boiling Cauldron', "event,Alchemist's Cookbook Base", 'Luck Charm', false, 'none'],
      crimson_cheese: ['Crimson Cheese', 'best.weapon.shadow', 'event,best.base.combo', 'Luck Charm', true, 'none']
    }
  };
  const settings = isProd
    ? getStorageToObject(targetEnvironmentType, defaultSetting)
    : defaultSetting;
  const quest = isProd ? user.quests.QuestMousoleum : QuestMousoleum;
  if (quest.can_build && settings.isAutoBuild)
    isProd
      ? hg.views.HeadsUpDisplayMousoleumView.buildWall()
      : console.log('buildWall');
  if (quest.can_repair && quest.wall_health < getRandomInteger(15, 25))
    isProd
      ? hg.views.HeadsUpDisplayMousoleumView.repairWall()
      : console.log('repairWall');
  let trap = settings.trap.radioactive_blue_cheese;
  if (quest.has_wall) {
    trap = settings.trap.radioactive_blue_cheese_wall;
    if (user.bait_item_id === 2700) {
      trap = settings.trap.crimson_cheese;
      const baitQuantity = parseQuantity(user.bait_quantity);
      if (baitQuantity < 3) trap = settings.trap.radioactive_blue_cheese_wall;
    }
  }
  if (trap[2].indexOf('event,') === 0)
    if (
      user.quests.QuestSpringHunt &&
      !user.quests.QuestSpringHunt.is_shutdown_enabled
    )
      trap[2] = 'Chocolate Bar Base';
    else trap[2] = trap[2].replace('event,');
  armAllTraps(...trap);
}
