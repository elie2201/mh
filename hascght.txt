function hasCaughtAction(isByType, check) {
  // const isProd = false;
  logging('hasCaughtAction_1.0.2.1.0, isByType', isByType, check);
  if (getEnvironmentType() === check.atEnvironmentType)
    if (
      (isByType && hasCaughtTargetByDataMouseType(check.target)) ||
      (!isByType && hasCaughtTargetByMouseName(check.target))
    ) {
      if (check.successCallback) {
        check.successCallback();
        return;
      }
    }
  if (check.beforeSuccessCallback) check.beforeSuccessCallback();
}
