  return true;

  console.log(`checkThenArm: ${sort}, ${category}, ${name}, ${isForce}`);
}

class CatchHandler {
  static checkedProperty = { isByMouseType: true, isByMouseName: false };
  static async run(fnName, ...args) {
    if (!fnName) return false;
    if (!Object.hasOwn(CatchHandler, fnName)) return false;
    if (typeof CatchHandler[fnName] !== 'function') return false;
    try {
      return await CatchHandler[fnName](...args);
    } catch (err) {
      console.error(`Error in ${fnName}:`, err);
      return false;
    }
  }
  /**
   * check journal to find if caught target mouse(by data-mouse-type/mouse name)
   * @param {object} checks
   * @returns
   */
  static async hasCaughtAction(checks) {
    logging('hasCaughtAction_2.0.0.0.0', checks);
    for (const check of checks) {
      if (MhUtils.getEnvironmentType() !== check.atEnvironmentType) continue;
      // 找到第一個 atEnvironmentType match的 check就終止迴圈
      console.log(`Is by data mouse type? ${check.isByMouseType}`);
      if (
        (check.isByMouseType && hasCaughtTargetByDataMouseType(check.target)) ||
        (!check.isByMouseType && hasCaughtTargetByMouseName(check.target))
      ) {
        if (check.successCallback) await check.successCallback();
        return true;
      }
      // 沒抓到目標老鼠時執行 beforeSuccessCallback
      if (check.beforeSuccessCallback) await check.beforeSuccessCallback();
      return false;
    }
    return false;
  }
  // Fiery Warpath check Warmonger
  static async checkWarmonger(args) {
    const checks = [
      {
        isByMouseType: CatchHandler.checkedProperty.isByMouseType,
        atEnvironmentType: 'desert_warpath',
        target: 'desert_boss',
        async successCallback() {
          MhUtils.travel('desert_city', 3000, () => {
            checkThenArm(null, 'bait', 'Gouda Cheese');
            checkThenArm(
              null,
              'weapon',
              MhUtils.parseTrapName('best.weapon.physical')
            );
            checkThenArm(
              null,
              'base',
              MhUtils.parseTrapName('best.base.combo')
            );
            checkThenArm(null, 'trinket', ['Artisan Charm', '2025 Charm']);
          });
        }
      }
    ];
    return await CatchHandler.hasCaughtAction(checks);
  }
}

  console.log(result); // true: hasCaught/false: not hasCaught
})();

/** check journal to find if caught target mouse(by data-mouse-type/mouse name) */
function hasCaughtAction(checks) {
  //* const isProd = false;
  logging('hasCaughtAction_1.1.2.1.3', checks);
  for (const check of checks) {
    if (MhUtils.getEnvironmentType() !== check.atEnvironmentType) continue;
    console.log(`Is by data mouse type? ${check.isByMouseType}`);
    if (
      (check.isByMouseType && hasCaughtTargetByDataMouseType(check.target)) ||
      (!check.isByMouseType && hasCaughtTargetByMouseName(check.target))
    ) {
      if (check.successCallback) {
        check.successCallback();
        return;
      }
    }
    if (check.beforeSuccessCallback) check.beforeSuccessCallback();
  }
}
