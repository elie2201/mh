/** check journal to find if caught target mouse(by data-mouse-type/mouse name) */
function hasCaughtAction(checks) {
  //* const isProd = false;
  logging('hasCaughtAction_1.1.2.1.3', checks);
  for (const check of checks) {
    if (MhUtils.getEnvironmentType() !== check.atEnvironmentType) continue;
    console.log(`Is by data mouse type? ${check.isByMouseType}`);
    if (
      (check.isByMouseType && hasCaughtTargetByDataMouseType(check.target)) ||
      (!check.isByMouseType && hasCaughtTargetByMouseName(check.target))
    ) {
      if (check.successCallback) {
        check.successCallback();
        return;
      }
    }
    if (check.beforeSuccessCallback) check.beforeSuccessCallback();
  }
}
