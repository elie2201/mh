// valourRift
if (getEnvironmentType() === 'rift_valour') valourRift();
function valourRift() {
  const valourRiftType = 'rift_valour';
  const environmentType = getEnvironmentType();
  if (environmentType != valourRiftType) return;
  logging('valourRift_0.1.10.2.1');
  // trap: [<bait>, <weapon>, <base>, <trinket>, <championFireEnabled>, litCandle, doubleEgg]
  // prettier-ignore
  const defaultSetting = {
    isAutoEnter: false,
    minGscToEnter: 200,
    isAutoRetreat: true,
    minRetreatFloor: 3,
    catchRate: 95,
    forceRetreatFloor: null,
    isAutoUpgrade: true,
    minCheeseQty: 20,
    buyCheeseQty: 55,
    trap: {
      farming: {
        0: ['Gauntlet String,Brie String', 'best.weapon.rift', 'best.base.rift', 'Rift Luck Charm,Rift Vacuum Charm', false, 'none', false]
      },
      tower: {
        0: ['Gauntlet String', 'best.weapon.rift', 'best.base.rift', 'Rift 2024 Charm,Rift 2025 Charm', false, 'none', false],
        8: ['Gauntlet String', 'best.weapon.rift', 'best.base.rift', 'Rift 2024 Charm,Rift 2025 Charm', false, 'none', false],
        16: ['Gauntlet String', 'best.weapon.rift', 'best.base.rift', 'Rift 2024 Charm,Rift 2025 Charm', false, 'none', false],
        24: ['Gauntlet String', 'best.weapon.rift', 'best.base.rift', 'Rift 2024 Charm,Rift 2025 Charm', false, 'none', false],
        eclipse: ['Gauntlet String', 'best.weapon.rift', 'best.base.rift', 'Rift 2024 Charm,Rift 2025 Charm', true, 'none', false]
      },
      uu: {
        0: ['Gauntlet String', 'best.weapon.rift', 'best.base.rift', 'Rift 2024 Charm,Rift 2025 Charm', false, 'none', false],
        8: ['Gauntlet String', 'best.weapon.rift', 'best.base.rift', 'Rift 2024 Charm,Rift 2025 Charm', false, 'none', false],
        16: ['Gauntlet String', 'best.weapon.rift', 'best.base.rift', 'Rift 2024 Charm,Rift 2025 Charm', false, 'none', false],
        24: ['Gauntlet String', 'best.weapon.rift', 'best.base.rift', 'Rift 2024 Charm,Rift 2025 Charm', false, 'none', false],
        eclipse: ['Gauntlet String', 'best.weapon.rift', 'best.base.rift', 'Rift 2024 Charm,Rift 2025 Charm', true, 'none', false]
      }
    }
  };
  const settings = getStorageToObject(valourRiftType, defaultSetting);
  const quest = user.quests.QuestRiftValour;
  const vriftStates = { farming: 'farming', tower: 'tower' };
  const state = quest.state;
  const floor = quest.floor;
  // const huntsLeft = quest.hunts_remaining;
  const powerUpData = quest.power_up_data;
  const speed = powerUpData.long_stride;
  const sync = powerUpData.hunt_limit;
  const siphon = powerUpData.boss_extension;
  const upgradeTypes = [
    'upgrade_long_stride',
    'upgrade_hunt_limit',
    'upgrade_boss_extension'
  ];
  const canUpgrade = [speed.can_upgrade, sync.can_upgrade, siphon.can_upgrade];
  const currentLevels = [
    speed.current_level + 1,
    sync.current_level + 1,
    siphon.current_level + 1
  ];
  console.log(canUpgrade, currentLevels);
  const autoRetreat = () => {
    // 先不處理 UU Run
    if (quest.is_eclipse_mode) return;
    // 針對那些接近但理論上到不了 eclipse floor的 upgrade,
    // 萬一運氣好到了 eclipse floor就撤退,不要浪費 CF賭 Eclipse
    const retreatUpgrades = ['[3,1,1]', '[5,3,3]', '[6,5,4]'];
    const retreatFloor = [8, 16, 24];
    const index = retreatUpgrades.indexOf(JSON.stringify(currentLevels));
    if (index > -1) settings.forceRetreatFloor = retreatFloor[index];
  };
  if (state === vriftStates.tower && floor >= settings.minRetreatFloor)
    autoRetreat();
  // claimCache();
  const claimCache = (callback) => {
    const element = '<a/>';
    hg.views.HeadsUpDisplayRiftValourView.riftValourAjax(
      element,
      { action: 'claim_cache' },
      callback
    );
  };
  if (quest.can_claim) {
    claimCache(() => {
      // document.querySelector('a.jsDialogClose, a#jsDialogClose').click();
      setTimeout(() => {
        window.location.href = MOUSEHUNT_BASE_URL;
      }, 1000);
    });
  }
  // retreat();
  const retreat = () => {
    console.log('retreat');
    const element = '<a/>';
    hg.views.HeadsUpDisplayRiftValourView.riftValourAjax(
      element,
      { action: 'retreat' },
      () => {
        claimCache(() => {
          // document.querySelector('a.jsDialogClose, a#jsDialogClose').click();
          setTimeout(() => {
            window.location.href = MOUSEHUNT_BASE_URL;
          }, 1000);
        });
      }
    );
  };
  if (state === vriftStates.tower && settings.isAutoRetreat) {
    if (settings.forceRetreatFloor && settings.forceRetreatFloor === floor) {
      retreat();
      return;
    }
  }
  const gscQuantity = parseQuantity(
    document.querySelector('div[data-item-type="gauntlet_string_cheese"]')
      .innerText
  );
  const elixirQuantity = parseQuantity(
    document.querySelector('div[data-item-type="gauntlet_elixir_stat_item"]')
      .innerText
  );
  if (state === vriftStates.farming && settings.isAutoEnter) {
    if (gscQuantity + elixirQuantity >= settings.minGscToEnter) {
      hg.views.HeadsUpDisplayRiftValourView.enterTower();
      setTimeout(() => {
        valourRift();
      }, 3000);
      return;
    }
  }
  const isAtEclipse = quest.is_at_eclipse;
  const trapKey = quest.is_eclipse_mode ? 'uu' : state;
  const floorSegments = [24, 16, 8];
  let floorKey = 0;
  if (state !== vriftStates.farming)
    for (let i = 0; i < floorSegments.length; i++) {
      const floorSegment = floorSegments[i];
      if (floor > floorSegment) {
        floorKey = floorSegment;
        break;
      }
    }
  let trap = settings.trap[trapKey][floorKey];
  if (isAtEclipse) trap = settings.trap[trapKey].eclipse;
  /* if (isAtEclipse || trap[4]) toggleChampionFire(true);
  else toggleChampionFire(false); */
  toggleChampionFire(trap[4]);
  setTimeout(() => {
    litCandle(trap[5]);
    doubleEgg(trap[6]);
  }, 500);
  setTimeout(() => {
    armAllTraps(...trap);
  }, 1500);
  // checkUpgrade();
  const checkUpgrade = () => {
    console.log('check upgrade');
    const upgradePlans = [
      [1, 1, 1],
      [2, 1, 1],
      [2, 2, 1],
      [3, 2, 1],
      [3, 3, 1],
      [3, 3, 2],
      [4, 3, 2],
      [5, 3, 2],
      [5, 3, 3],
      [6, 3, 3],
      [6, 3, 4],
      [6, 4, 4],
      [6, 5, 4],
      [6, 6, 4],
      [7, 6, 4],
      [8, 6, 4],
      [8, 6, 5],
      [9, 6, 5],
      [10, 6, 5],
      [10, 7, 5]
    ];
    if (canUpgrade.indexOf(true) < 0) return;
    for (let i = 0; i < upgradePlans.length; i++) {
      const plan = upgradePlans[i];
      for (let j = 0; j < plan.length; j++) {
        const level = plan[j];
        console.log(
          `upgrade ${upgradeTypes[j]} from level ${
            currentLevels[j]
          } to ${level}? ${canUpgrade[j] === true}`
        );
        if (currentLevels[j] < level) {
          console.log(
            `find goal ${upgradeTypes[j]}, from level${currentLevels[j]} to level${level}`
          );
          if (canUpgrade[j] === true)
            hg.views.HeadsUpDisplayRiftValourView.riftValourAjax('<a/>', {
              action: upgradeTypes[j]
            });
          return;
        }
      }
    }
  };
  if (settings.isAutoUpgrade)
    setTimeout(() => {
      checkUpgrade();
    }, 6000);
  const autoBuy = () => {
    logging('autoBuy');
    if (gscQuantity < getRandomInteger(14, 23)) {
      logging('auto buy Gauntlet');
      const itemType = 'gauntlet_string_cheese';
      const itemQuantity = Math.min(elixirQuantity, 55);
      completeTransaction(true, itemType, itemQuantity, false);
    }
    if (
      user.trinket_item_id === 1553 &&
      parseQuantity(user.trinket_quantity) < 22
    ) {
      logging('auto buy Rift Vacuum');
      const itemType = 'rift_vacuum_trinket';
      const itemQuantity = 222;
      setTimeout(() => {
        completeTransaction(true, itemType, itemQuantity, false);
      }, 3000);
    }
  };
  setTimeout(() => {
    autoBuy();
  }, 10000);
}
function toggleChampionFire(isToEnable) {
  logging(`toggleChampionFire to ${isToEnable}`);
  const quest = user.quests.QuestRiftValour;
  const itemQuantity = parseQuantity(
    document.querySelector('div[data-item-type="rift_gauntlet_fuel_stat_item"]')
      .innerText
  );
  if (itemQuantity < 1) return;
  const isFuelEnabled = quest.is_fuel_enabled || false;
  if ((isFuelEnabled === true) != isToEnable)
    hg.views.HeadsUpDisplayRiftValourView.toggleFuel();
}
