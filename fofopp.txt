function getStorageToObject(keyName, objDefault) {
  let obj = getStorage(keyName);
  let bCheckNewProp = true;
  if (isNullOrUndefined(obj)) {
    obj = JSON.stringify(objDefault);
    setStorage(keyName, obj);
    bCheckNewProp = false;
  }
  obj = JSON.parse(obj);
  if (bCheckNewProp) {
    if (assignMissingDefault(obj, objDefault)) {
      setStorage(keyName, JSON.stringify(obj));
    }
  }
  return obj;
  function getStorage(name) {
    // check if the web browser support HTML5 storage
    if ('localStorage' in window && !isNullOrUndefined(window.localStorage)) {
      return window.localStorage.getItem(name);
    }
  }
  function setStorage(name, value) {
    // check if the web browser support HTML5 storage
    if ('localStorage' in window && !isNullOrUndefined(window.localStorage)) {
      window.localStorage.setItem(name, value);
    }
  }
  function isNullOrUndefined(obj) {
    return (
      obj === null || obj === undefined || obj === 'null' || obj === 'undefined'
    );
  }
  function assignMissingDefault(target, source) {
    let isResave = false;
    for (const key in source) {
      if (
        Object.prototype.hasOwnProperty.call(source, key) &&
        !Object.prototype.hasOwnProperty.call(target, key)
      ) {
        const element = source[key];
        target[key] = element;
        isResave = true;
      }
    }
    return isResave;
  }
}

            // 需要買嗎?如果該買了, itemQuantity一定大於 0
            // itemQuantity > 0 &&
            // clam/gold夠買嗎?
            itemQuantity > settings.useBaitQuantity - items.clamembert.qty
          ) {
            console.log(`auto buy ${itemQuantity} ${items.clamembert.type}`);
            completeTransaction(
              true,
              items.clamembert.type,
              itemQuantity,
              false,
              () => {
                items.clam.qty -= itemQuantity;
                items.clamembert.qty += itemQuantity;
                this.autoBuyPromise();
              }
            );
            return;
          }
          // 能買多少 chum?
          itemQuantity = Math.min(
            items.penny.qty - settings.extraPenny,
            Math.floor(items.gold / 1000)
          );
          if (
            // 要買跟 clamembert等量的 chum嗎?
            items.chum.qty < items.clamembert.qty &&
            // 需要買嗎?如果該買了, itemQuantity一定大於 0
            // itemQuantity > 0 &&
            // penny/gold夠買嗎?
            itemQuantity > items.clamembert.qty - items.chum.qty
          ) {
            console.log(`auto buy ${itemQuantity} ${items.chum.type}`);
            completeTransaction(
              true,
              items.chum.type,
              itemQuantity,
              false,
              () => {
                items.penny.qty -= itemQuantity;
                items.chum.qty += itemQuantity;
                this.autoBuyPromise();
              }
            );
            return;
          }
          // 能買多少 grubbeen?
          itemQuantity = Math.min(
            items.grub.qty,
            Math.floor(items.gold / 1000)
          );
          // grubbeen: with chum=>clam, without=>penny
          if (
            // 該買 grubbeen了沒?
            items.grubbeen.qty < settings.changeBaitQuantity &&
            // grub/gold夠買嗎?
            itemQuantity > settings.useBaitQuantity - items.grubbeen.qty
          ) {
            console.log(`auto buy ${itemQuantity} ${items.grubbeen.type}`);
            completeTransaction(
              true,
              items.grubbeen.type,
              itemQuantity,
              false,
              () => {
                items.grub.qty -= itemQuantity;
                items.grubbeen.qty += itemQuantity;
                this.autoBuyPromise();
              }
            );
            return;
          }
          // 能買多少 chum?
          itemQuantity = Math.min(
            items.penny.qty - settings.extraPenny,
            Math.floor(items.gold / 1000)
          );
          if (
            // 要買跟 grubbeen等量的 chum嗎?
            items.chum.qty < items.grubbeen.qty &&
            // 需要買嗎?如果該買了, itemQuantity一定大於 0
            // itemQuantity > 0 &&
            // penny/gold夠買嗎?
            itemQuantity > items.grubbeen.qty - items.chum.qty
          ) {
            console.log(`auto buy ${itemQuantity} ${items.chum.type}`);
            completeTransaction(
              true,
              items.chum.type,
              itemQuantity,
              false,
              () => {
                items.penny.qty -= itemQuantity;
                items.chum.qty += itemQuantity;
                this.autoBuyPromise();
              }
            );
            return;
          }
          this.autoSettingsPromise();
        });
      },
      /** auto settings */
      autoSettingsPromise() {
        logging('run initPromise: initialize states for running');
        return new Promise((resolve, reject) => {
          // auto settings for running if necessary and then invoke next promise
          // trapState, default: clamembert
          if (
            settings.isOkStormy &&
            items.stormy.qty > settings.useBaitQuantity
          ) {
            this.trapState = stormy;
          } else if (
            !settings.isOkStormy ||
            items.stormy.qty < settings.changeBaitQuantity
          ) {
            if (items.clamembert.qty >= settings.useBaitQuantity) {
              this.trapState = clamembert;
              if (items.chum.qty >= settings.useBaitQuantity)
                settings.isOkChum = true;
            } else if (items.clamembert.qty < settings.changeBaitQuantity) {
              if (items.grubbeen.qty >= settings.useBaitQuantity) {
                this.trapState = grubbeen;
                if (items.chum.qty >= settings.useBaitQuantity)
                  settings.isOkChum = true;
              } else if (items.grubbeen.qty < settings.changeBaitQuantity) {
                this.trapState = gouda;
              }
            }
          }
          this.handleTrapPromise();
        });
      },
      /** handle trap promise */
      handleTrapPromise() {
        logging('run handleTrapPromise');
        return new Promise((resolve, reject) => {
          const trap = settings.traps[this.trapState];
          if (trap) {
            this.toggleCC(trap[4]);
            setTimeout(() => {
              litCandle(trap[5]);
              doubleEgg(trap[6]);
            }, 500);
            setTimeout(() => {
              if (items.chum.qty > 0 && settings.isOkChum)
                isProd
                  ? hg.views.HeadsUpDisplayProloguePondView.toggleChum('<a/>')
                  : console.log('scatter chum');
              armAllTraps(...trap);
            }, 1500);
          }
        });
      },
      /** toggle condensed creativity */
      toggleCC(isToEnable) {
        logging(`toggleCC to ${isToEnable}`);
        if (items.CC.qty < 1) return;
        const isFuelEnabled = quest.is_fuel_enabled || false;
        if ((isFuelEnabled === true) != isToEnable)
          isProd
            ? hg.views.HeadsUpDisplayFolkloreForestRegionView.toggleFuel()
            : console.log('toggled CC');
      },
      /** location ajax */
      ajax(element, params, successCallback, errorCallback) {
        params['last_read_journal_entry_id'] = $('#journallatestentry').data(
          'entry-id'
        );
        params['uh'] = user.unique_hash;
        $.ajax({
          data: params,
          dataType: 'json',
          type: isProd ? 'post' : 'get',
          url: isProd
            ? MOUSEHUNT_BASE_URL + this.URI
            : 'https://elie2201.github.io/mh/jobs.txt',
          success: function (data) {
            eventRegistry.doEvent('ajax_response', data);
            if (successCallback) {
              successCallback(data);
            }
          },
          error: function (data) {
            // eslint-disable-next-line new-cap
            const popup = new jsDialog();
            popup.setTemplate('error');
            popup.addToken('{*title*}', 'An error has occurred.');
            popup.addToken(
              '{*content*}',
              'An error occurred. Please try again!'
            );
            popup.show();
            if (errorCallback) errorCallback(data);
          }
        });
      }
    };
    // initial utils
    function init() {
      // make fields immutable
      utils.setImmutable();
      // initialize values in utils if necessary
    }
    init();
    return utils;
  })();
  prologuePondUtility.run();
}
