  function skyMap(islands, settings, isHai) {
    const isSkyMapLaunched = true; // return true to stop invoker(from reloading/trap arming in launch)
    const myPowerTypes = isHai
      ? settings.myHaiPowerTypes
      : settings.myLaiPowerTypes;
    const shrineKey = isHai ? 'paragon_cache' : 'shrine';
    const included = settings.includedHaiShrine;
    const choices = [];
    // 防呆:測試時曾經忘記放 shrineMaxIndex導致無限 cycle
    const shrineMaxIndex = settings.shrineMaxIndex
      ? settings.shrineMaxIndex
      : 2;
    for (let i = 0; i < shrineMaxIndex; i++) {
      for (const myPowerType of myPowerTypes) {
        const modifier = islands[myPowerType][i];
        //* if (modifier.indexOf('shrine') > -1) {
        if (modifier?.includes(shrineKey)) {
          const isFilterOk =
            !included ||
            included.length == 0 || // 防呆: included中 4個 shrine都沒有的話會無限 cycle
            included.includes(modifier);
          console.log(`isHaiOk? ${isFilterOk}`);
          if (!isHai || isFilterOk) choices.push(myPowerType);
        }
      }
    }
    if (choices.length > 0) {
      const choice = bestChoice(islands, choices, settings.wantedModifier);
      console.plog('launchSkyMap', choice, choices, islands);
      if (choice) {
        launchSkyMap(choice); // TODO 測試時 comment掉
        return isSkyMapLaunched;
      }
    }
    // Use Cyclone or choose third-tile island?
    // If use cyclone: terminate running autoSkyMap > confirm cycled > call treatSkyMap in cycled again
    // TODO 不測試 cycled()時用這行 const isCycle = false;
    const isCycle =
      (isHai ? settings.isCycleHai : settings.isCycleLai) && cycloneQty > 0;
    if (isCycle) {
      cycled(settings);
      return isSkyMapLaunched;
    }
    // 到這邊的話 choices為空且不 cycle,直接塞第一個 powerType
    launchSkyMap(myPowerTypes[0]);
    return isSkyMapLaunched;
  }
  function cycled(settings) {
    const getTileTypes = () => {
      const tiles = $('div[data-type]', $('#overlayPopup'));
      const tileTypes = [];
      tiles.each((i, e) => {
        tileTypes.push(e.getAttribute('data-type'));
      });
      return tileTypes;
    };
    const sameArray = (a, a1) => {
      for (let i = 0; i < a.length; i++) {
        const element = a[i];
        if (!a1[i] || a1[i] !== element) return false;
      }
      return true;
    };
    const tileTypes = getTileTypes();
    const rerollClass = '.floatingIslandsAdventureBoardSkyMap-rerollButton';
    //* document.querySelector(rerollClass).click();
    document.querySelector(rerollClass).dispatchEvent(new Event('click'));
    const key = setInterval(() => {
      const newTileTypes = getTileTypes();
      if (sameArray(newTileTypes, tileTypes)) {
        return;
      }
      clearInterval(key);
      treatSkyMap(settings);
    }, 1000);
  }
  function bestChoice(islands, choices, wantedModifier) {
    // TODO Using wantedModifier
    for (const choice of choices) {
      const island = islands[choice];
      if (island.filter((elem) => elem === 'loot_cache').length === 2)
        return choice;
    }
    for (const choice of choices) {
      const island = islands[choice];
      if (island.filter((elem) => elem === 'sky_pirates').length === 2)
        return choice;
    }
    // Lai的話 shrine first優先, Hai的話這個判斷無作用
    for (const choice of choices) {
      const island = islands[choice];
      if (island[0].includes('shrine')) return choice;
    }
    for (const choice of choices) {
      const island = islands[choice];
      //* const count = island.filter(elem => elem === 'empty_sky').length;
      if (island.includes('loot_cache') && !island.includes('empty_sky'))
        return choice;
    }
    for (const choice of choices) {
      const island = islands[choice];
      if (!island.includes('empty_sky') && distinctArray(island).length === 2)
        return choice;
    }
    for (const choice of choices) {
      const island = islands[choice];
      //* const count = island.filter(elem => elem === 'empty_sky').length;
      if (island.includes('loot_cache')) return choice;
      if (island.includes('sky_pirates')) return choice;
      if (!island.includes('empty_sky')) return choice;
    }
    return choices[0];
    //* return null;
  }
