/** Utilities */
const MyUtils = (function () {
  const utils = {
    /**
     * logging function.
     * Invoke console.log with customization.
     * @param  {...any} logs
     */
    logging(...any) {
      if (isDebug) {
        console.groupCollapsed(new Date(), ...any);
        // console.log(Error().stack);
        console.trace(...any);
        console.groupEnd();
      }
    },
    /**
     * Format date to ISO date string format yyyy-MM-dd.
     * Return today's date string if no argument
     */
    toISODateString(date) {
      const d = date ? date : new Date();
      return d.toLocaleDateString('en-CA');
    },
    /** Remove any comma in number expression and then parse to integer
     * return 0 if argument is null or isNaN(after remove comma)
     */
    parseQuantity(qty) {
      const strQty = ((qty || 0) + '').trim();
      const rtn = parseInt(strQty.replaceAll(',', ''));
      return Number.isNaN(rtn) ? 0 : rtn;
    },
    /** trim string or number, to empty if null or undefined */
    trimToEmpty(str) {
      return this.trimToDefault(str);
    },
    /** trim string or number, to default if null or undefined */
    trimToDefault(str, deflt) {
      return str === 0 ? '0' : ((str || (deflt ? deflt : '')) + '').trim();
    },
    /** get a integer between min and max */
    getRandomInteger(min, max) {
      return parseInt(min + Math.random() * (max - min));
    },
    /** create new array containing distinct element of original array */
    distinctArray(array) {
      const rtn = [];
      for (let i = 0; i < array.length; i++) {
        const element = array[i];
        if (rtn.indexOf(element) < 0) rtn.push(element);
      }
      return rtn;
    },
    /**
     * Check if argument is null or undefined
     * or string of both.
     * Not including empty string.
     * @param {*} obj
     * @return true if argument is null
     * or undefined or string of both.
     */
    isNullOrUndefined(obj) {
      return (
        obj === null ||
        obj === undefined ||
        obj === 'null' ||
        obj === 'undefined'
      );
    },
    /**
     * 用 keyName從 localStorage getItem.
     * 如果取出來的字串 isNullOrUndefined,
     * 將 objDefault stringify存回 localStorage.
     * json parse取出的字串成物件 obj.
     * 將 obj與 objDefault比對屬性,
     * obj如果有缺少 objDefault的屬性,
     * 將缺少的屬性從 objDefault copy給obj後回寫 localStorage.
     * @param {String} keyName
     * @param {*} objDefault
     * @return {*} json parse後的物件
     */
    getStorageToObject(keyName, objDefault) {
      let obj = this.getStorage(keyName);
      let bCheckNewProp = true;
      if (this.isNullOrUndefined(obj)) {
        obj = JSON.stringify(objDefault);
        this.setStorage(keyName, obj);
        bCheckNewProp = false;
      }
      obj = JSON.parse(obj);
      if (bCheckNewProp) {
        if (this.assignMissingDefault(obj, objDefault)) {
          this.setStorage(keyName, JSON.stringify(obj));
        }
      }
      return obj;
    },
    /**
     * if the web browser support HTML5 storage,
     * 用 name作為 key從 localStorage中 getItem.
     * @param {String} name key name
     * @return
     */
    getStorage(name) {
      if (
        'localStorage' in window &&
        !this.isNullOrUndefined(window.localStorage)
      ) {
        return window.localStorage.getItem(name);
      }
    },
    /** if the web browser support HTML5 storage, store it */
    setStorage(name, value) {
      if (
        'localStorage' in window &&
        !this.isNullOrUndefined(window.localStorage)
      ) {
        window.localStorage.setItem(name, value);
      }
    },
    /**
     * 將 objDefault中有但 obj沒有的屬性 copy給 obj
     * @param {*} obj
     * @param {*} defaultObj
     * @return {Boolean} if any property copied
     */
    assignMissingDefault(obj, defaultObj) {
      let isResave = false;
      for (const key in defaultObj) {
        if (
          Object.prototype.hasOwnProperty.call(defaultObj, key) &&
          !Object.prototype.hasOwnProperty.call(obj, key)
        ) {
          const element = defaultObj[key];
          obj[key] = element;
          isResave = true;
        }
      }
      return isResave;
    },
    /**
     * Parse datetime or time str to datetime in milliseconds.
     * Prepend today's Date if str is time expression(without T or start with T).
     * Express UTC time with Z at end.
     */
    parseDateTime(str) {
      if (this.trimToEmpty(str) === '') return null;
      if (str.indexOf('T') === 0)
        return Date.parse(this.toISODateString() + str);
      if (str.indexOf('T') > 0) return Date.parse(str);
      return Date.parse(this.toISODateString() + 'T' + str);
    }
  };
  return utils;
})();

  const mappingUtility = (() => {
    const utils = {
      // values

      /** Set immutable fields */
      setImmutable() {
        const immutableFields = ['URI'];
        for (let i = 0; i < immutableFields.length; i++) {
          const immutableField = immutableFields[i];
          Object.defineProperty(utils, immutableField, {
            // value: 'This cannot be changed',
            writable: false, // Makes the field immutable
            configurable: false, // Prevents deletion or re-configuration of the field
            enumerable: true // Makes the field visible during enumeration (e.g., in a for...in loop)

      snuid: null,
      mapTypeKey: null,
      mapInfoParser: null,
      callback: null,

      /** start running */
      run(callback) {
        const version = 'start run mappingUtility';
        isProd ? logging(version) : console.log(version);
        if (!mapTypeKey) return console.log('No working map');
        this.callback = callback;
        this.autoSettingsPromise().then(() => {
          this.autoMapping(this.mapTypeKey, this.callback);
        });
      },
      /** auto settings */
      autoSettingsPromise() {
        const version = 'start autoSettingsPromise';
        isProd ? logging(version) : console.log(version);
        return new Promise((resolve, reject) => {
          // auto settings if necessary and then invoke next promise
          this.scrollChest = scrollChests[this.mapTypeKey];
          this.map = MhUtils.getMapByClass(this.scrollChest.mapClass);
          resolve();
        });
      },
      /** autoMapping */
      autoMapping() {
        const version = 'autoMapping_1.0.1.0.4';
        isProd ? logging(version) : console.log(version);
        const selfInvoking = () => {
          setTimeout(() => {
            utils.autoMapping();
          }, 1000);
        };
        if (!this.scrollChest) {
          if (this.callback) this.callback();
          return;
        }
        // autoMapping會 self invoking,必須自行更新 this.map
        this.map = MhUtils.getMapByClass(this.scrollChest.mapClass);
        console.log('claim map reward checking');
        if (this.map && this.map.is_complete) {
          // clear this.map in claimAllMapReward
          this.claimAllMapReward(selfInvoking);
          return;
        }
        console.log('accept invite checking');
        const quest = user.quests.QuestRelicHunter;
        if (!this.map && (quest.num_invites || 0) > 0) {
          this.acceptFirstInvite(this.scrollChest.mapClass, selfInvoking);
          return;
        }
        const okMaptain = [maptainTypes.always, maptainTypes.once];
        console.log('autoMaptain: buy scroll=>use scroll=>(invite) checking');
        if (!this.map) {
          if (okMaptain.indexOf(localStorage[maptainTypeKey]) > -1) {
            console.log('is autoMaptain, buy=>use scroll=>invite');
            this.autoMaptain(
              this.scrollChest.size,
              this.scrollChest.mapClass,
              this.scrollChest.scroll,
              selfInvoking
            );
            return;
          }
          console.log('不在地圖中又不是 maptain,結束 autoMapping呼叫 callback');
          this.callback();
          return;
        }
        console.log('Must be in map, mapId:', this.map.map_id);
        hg.utils.TreasureMapUtil.getMapInfo(
          this.map.map_id,
          (data) => {
            const info = data; // isProd ? data : MapInfo;
            utils.mapInfo = parseMapInfo(info);
            if (!utils.mapInfo) {
              utils.map = null;
              utils.callback();
              return;
            }
            if (utils.mapInfo && utils.mapInfo.isCompleted) {
              const msg = '到這裡不應該 map is completed.game data沒 sync?';
              isProd ? console.plog(msg) : console.log(msg);
              setTimeout(() => {
                MhUtils.reloadWithMessage('Why map completed!? Reloading...');
                // run();
              }, 2000);
              return;
            }
            // 先判斷 mapInfo,減少讀取 localStorage
            console.log('mapInfo exists and is full?', utils.mapInfo);
            if (utils.mapInfo && utils.mapInfo.isFull) {
              if (!utils.mapInfo.isDusted) {
                console.log('Map is full and not dusted, ok to dust');
                if (
                  okMaptain.indexOf(localStorage[dusterCycleKey]) > -1 &&
                  this.snuid === utils.mapInfo.maptain
                ) {
                  console.log('I am maptain and ok to dust');
                  utils.autoDust(() => {
                    utils.mapInfo.isDusted = true;
                    console.log(
                      `localStorage: maptainType:${localStorage[maptainTypeKey]},dusterCycle:${localStorage[dusterCycleKey]}`
                    );
                    utils.callback();
                  });
                  return;
                }
              }
              console.log('Map is full and dusted, auto next maptain');

                `localStorage: maptainType:${localStorage[maptainTypeKey]},dusterCycle:${localStorage[dusterCycleKey]}`
              );
              utils.callback();
              return;
            }
            utils.callback();
          },
          // 這邊必須有 mapInfo了,沒有就 reload
          () => {
            MhUtils.reloadWithMessage('getMapInfo failed, Reloading...');
          }
        );
      },
      /**
       * 根據 mapGroups設定, attendance(12)人 RR時,
       * group中的第 n人:
       * n = ((mapSize - (attendance % mapSize)) % mapSize) + 1,
       * index: (mapSize - (attendance % mapSize)) % mapSize,
       * 為下一個 maptain+duster.
       * maptainType/dusterCycle為 never時不自動 next maptain.
       */
      autoNextMaptain(attendance, mapSize, mapInfo) {
        const version = `autoNextMaptain_1.0.2.1.0(${attendance}, ${mapSize},`;
        isProd ? logging(version, mapInfo) : console.log(version, mapInfo);
        const mapGroup = this.findNowGroup(mapSize, mapInfo);
        if (!mapGroup) return;
        // 用 mapSize而非 mapGroup.length - 1,就可以避掉 solo的情況
        const nextMaptainIndex = mapSize; // (mapSize - (attendance % mapSize)) % mapSize;
        const isNextMaptain = this.snuid === mapGroup[nextMaptainIndex];
        console.log(`I am${isNextMaptain ? '' : ' not'} next maptain`);
        if (!isNextMaptain) return;
        // 判斷不是 once只是為了減少寫入 localStorage,沒有業務邏輯
        // 判斷不是 never才是新功能: 停止 loop-mapping
        const notWriteTypes = [maptainTypes.once, maptainTypes.never];
        if (notWriteTypes.indexOf(localStorage[maptainTypeKey]) < 0)
          localStorage[maptainTypeKey] = maptainTypes.once;
        if (notWriteTypes.indexOf(localStorage[dusterCycleKey]) < 0)
          localStorage[dusterCycleKey] = maptainTypes.once;
      },
      /** findNowGroup */
      findNowGroup(mapSize, mapInfo) {
        const version = 'findNowGroup(';
        isProd ? logging(version, mapInfo) : console.log(version, mapInfo);
        const maptain = mapInfo.maptain;
        const groups = mapGroups[mapSize];
        if (!groups) return null;
        for (let i = 0; i < groups.length; i++) {
          const mapGroup = groups[i];
          const groupMaptain = mapGroup[0];
          if (groupMaptain === maptain) {
            console.log('mapGroup found:', mapGroup);
            return mapGroup;
          }
        }
        return null;
      },
      /** autoDust */
      autoDust(callback) {
        const version = 'autoDust_1.0.0.0.0';
        isProd ? logging(version, callback) : console.log(version, callback);
        const dusterCycle = localStorage[dusterCycleKey];
        if ([maptainTypes.always, maptainTypes.once].indexOf(dusterCycle) < 0) {
          callback(); // autoMapping不須先檢查,異常就跳過跑 callback
          return;
        }
        if (!this.mapInfo) {
          callback(); // 需要 mapInfo的 mapId,沒有就跳過跑 callback
          return;
        }
        // upgrade map
        const next = () => {
          if (dusterCycle === maptainTypes.once)
            localStorage[dusterCycleKey] = maptainTypes.no;
          callback();
        };
        const mapId = this.mapInfo.mapId;
        if (mapId) hg.utils.TreasureMapUtil.upgrade(mapId, next, callback);
      },
      /** Claim "all" claimable map reward */
      claimAllMapReward(callback) {
        const version = 'claimAllMapReward_1.0.1.0.0';
        isProd ? logging(version) : console.log(version);
        const maps = user.quests.QuestRelicHunter.maps;
        // 沒 map視同完成 claim檢查,執行 callback離開自己
        if (!maps || maps.length == 0) {
          if (callback) callback();
          return;
        }
        const selfInvoking = () => {
          utils.claimAllMapReward(callback);
        };
        for (let i = 0; i < maps.length; i++) {
          const map = maps[i];
          console.log(
            'Check auto claim map reward!!',
            map.map_id,
            map.is_complete
          );
          if (map.is_complete) {
            const mapName = MyUtils.trimToEmpty(map.name);
            const isMapRare = map.is_rare;
            console.log(
              `Claim map reward, id: ${map.map_id}, name: ${mapName}`
            );
            // 先 claim reward
            hg.utils.TreasureMapUtil.claim(map.map_id, () => {
              const scrollChest = utils.findScrollChest(mapName);
              // const scrollChest = scrollChests[MyUtils.trimToEmpty(mapName)];
              // claim後如果有對應的 chest且自動開,等 1秒開箱
              if (scrollChest && scrollChest.isAutoOpen) {
                const chest = isMapRare
                  ? scrollChest['chestRare']
                  : scrollChest['chest'];
                setTimeout(() => {
                  hg.utils.UserInventory.useConvertible(chest, 1, (data) => {
                    console.log(data.success == 1 ? 'success' : 'failed', data);
                    // 開箱後回 call自己檢查還有沒有待 claim的 map
                    setTimeout(selfInvoking, 1000);
                  });
                }, 1000);
                return;
              }
              // 沒開箱的話,直接回 call自己檢查還有沒有待 claim的 map
              setTimeout(selfInvoking, 1000);
            });
            // 有處理到完成的 map,就中斷迴圈
            return;
          }
        }
        // for loop沒有被中斷並回 call的話:
        // utils.map也應該被 claim了
        utils.map = null;
        // 只剩不能 claim=完成 claim檢查, callback
        if (callback) callback();

      /** autoMaptain */
      autoMaptain(mapSize, mapClass, scrollType, callback) {
        const version = `autoMaptain(${mapSize}, ${mapClass}, ${scrollType})_1.0.2.0.0`;
        isProd ? logging(version) : console.log(version);
        const maptainType = localStorage[maptainTypeKey];
        if ([maptainTypes.always, maptainTypes.once].indexOf(maptainType) < 0)
          return; // autoMapping必須先檢查,不然異常,斷
        if (!mapClass || !MapClasses[mapClass]) return; // 異常,斷
        if (!scrollType || scrollType == '') return; // 異常,斷
        const map = MhUtils.getMapByClass(mapClass);
        if (map) return; // autoMapping必須先檢查,不然異常,斷
        const groups = mapGroups[mapSize];
        let snuids = null;
        for (let i = 0; i < groups.length; i++) {
          snuids = groups[i];
          if (snuids.indexOf(this.snuid) === 0) break;
        }
        // 下面那行很危險,直接 return程式就斷了
        // 如果 callback(回 autoMapping檢查沒有 map又是 maptain)絕對無限循環
        if (!snuids || !Array.isArray(snuids) || snuids.length == 0) {
          const msg = 'mapGroup not found';
          MhUtils.storeError(msg);
          return;
        }
        console.log('map group found, open scroll and invite', snuids);
        snuids = snuids
          .filter((value) => value !== this.snuid)
          .filter((value, index, array) => array.indexOf(value) === index);
        console.log('After remove self and duplicate', snuids, mapGroups);
        MhUtils.completeTransaction(true, scrollType, 1, false, () => {
          hg.utils.UserInventory.useConvertible(
            scrollType,
            1,
            () => {
              console.log('After scroll opened, invite map members');
              utils.autoInvite(mapClass, snuids, callback);
              if (maptainType === maptainTypes.once)
                localStorage[maptainTypeKey] = maptainTypes.no;
            },
            parseInt(5000 + Math.random() * 2000)
          );
        });
      },
      /** Auto invite as scroll opened */

            MhUtils.closeAnyJsDialog();
            if (callback) callback();
          }, 2000);
          return;
        }
        hg.utils.TreasureMapUtil.sendInvites(map.map_id, snuids);
        setTimeout(() => {
          MhUtils.closeAnyJsDialog();
          if (callback) callback();
        }, 2000);
      },
      /** location ajax */
      ajax(element, params, successCallback, errorCallback) {
        params['last_read_journal_entry_id'] = $('#journallatestentry').data(
          'entry-id'
        );
        params['uh'] = user.unique_hash;
        $.ajax({
          data: params,
          dataType: 'json',
          type: isProd ? 'post' : 'get',
          url: isProd
            ? MOUSEHUNT_BASE_URL + this.URI
            : 'https://elie2201.github.io/mh/jobs.txt',
          success: function (data) {
            eventRegistry.doEvent('ajax_response', data);
            if (successCallback) {
              successCallback(data);
            }
          },
          // prettier-ignore
          error: function (data) {
            // eslint-disable-next-line new-cap
            const popup = new jsDialog();
            popup.setTemplate('error');
            popup.addToken('{*title*}', 'An error has occurred.');
            popup.addToken('{*content*}', 'An error occurred. Please try again!');
            popup.show();
            if (errorCallback) errorCallback(data);
          }
        });

      const hunter = hunters[i];
      if (hunter.captain) rtn.maptain = (hunter.sn_user_id + '').trim();
      const completedIds = hunter.completed_goal_ids.mouse;
      if (completedIds && completedIds.length > 0)
        completed = completed.concat(completedIds);
    }
    console.log('completed mouse ids', completed);
    rtn.completed = completed;
    const mouses = myMap.goals.mouse;
    for (let i = 0; i < mouses.length; i++) {
      const mouse = mouses[i];
      if (completed.indexOf(mouse.unique_id) > -1) continue;
      const type = mouse.type;
      if (!rtn[all]) rtn[all] = [];
      rtn[all].push(type);
    }
    console.log('parsed mapInfo:', rtn);
