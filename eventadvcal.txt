function claimAdventCalendar() {
  const isProd = false;
  //   const targetEnvironmentType = 'fort_rox';
  //   const enviromentType = getEnvironmentType();
  if (!user.quests.MiniEventAdventCalendar) return;
  logging('claimAdventCalendar_1.0.0.0.0');
  // my own hard-coded constants
  /** localStorage settings */
  // prettier-ignore
  //   const defaultSettings = {};
  //   const settings = isProd
  //     ? getStorageToObject(targetEnvironmentType, defaultSettings)
  //     : defaultSettings;
  // MH constants
  const quest = isProd ? user.quests.MiniEventAdventCalendar : MiniEventAdventCalendar;
  //   const trapState = quest.mice_state;
  const claimAdventCalendarUtility = (() => {
    const utils = {
      // values
      /** ajax URI */
      URI: 'managers/ajax/events/advent_calendar.php',
      /** Set immutable fields */
      setImmutable() {
        const immutableFields = ['URI'];
        for (let i = 0; i < immutableFields.length; i++) {
          const immutableField = immutableFields[i];
          Object.defineProperty(utils, immutableField, {
            // value: 'This cannot be changed',
            writable: false, // Makes the field immutable
            configurable: false, // Prevents deletion or re-configuration of the field
            enumerable: true // Makes the field visible during enumeration (e.g., in a for...in loop)
          });
        }
      },
      /** start running */
      run() {
        logging('start claimAdventCalendar');
        this.autoSettingsPromise();
      },
      /** auto settings */
      autoSettingsPromise() {
        logging('start autoSettingsPromise');
        return new Promise((resolve, reject) => {
          // auto settings for running if necessary and then invoke next promise
          this.claimPromise();
        });
      },
      /** claim advent calendar promise */
      claimPromise() {
        logging('start claimPromise');
        return new Promise((resolve, reject) => {
          const gifts = quest.gifts;
          for (let i = 0; i < gifts.length; i++) {
            const gift = gifts[i];
            const giftBit = gift.gift;
            if (gift.can_claim && !gift.has_claimed) {
              console.log('can claim and not has claimed, try claiming');
              this.ajax(
                {
                  action: 'claim',
                  gift: giftBit
                },
                () => {
                  gift.can_claim = false;
                  gift.has_claimed = true;
                }
              );
              return;
            }
          }
          console.log('there is nothing to claim');
          //   const trap = settings.traps[trapState];
          //   if (trap) armAllTraps(...trap);
        });
      },
      /** location ajax */
      ajax(params, successCallback, errorCallback) {
        return new Promise((resolve, reject) => {
          const defaultParams = {
            last_read_journal_entry_id: $('#journallatestentry').data(
              'entry-id'
            ),
            uh: user.unique_hash
          };
          $.ajax({
            data: { ...defaultParams, ...params },
            dataType: 'json',
            type: 'post',
            url: isProd
              ? MOUSEHUNT_BASE_URL + this.URI
              : 'https://elie2201.github.io/mh/jobs.txt',
            success: function (data) {
              eventRegistry.doEvent('ajax_response', data);
              if (successCallback) successCallback();
              return resolve(data);
            },
            error: (error) => {
              if (errorCallback) errorCallback();
              return reject(error);
            }
          });
        });
      }
    };
    // initial utils
    function init() {
      // make fields immutable
      utils.setImmutable();
      // initialize values in utils if necessary
    }
    init();
    return utils;
  })();
  claimAdventCalendarUtility.run();
}
