function scheduledTravel(
  beginTime,
  endTime,
  environmentType,
  timeout,
  callback
) {
  const isProd = false;
  const version = `scheduledTravel_1.0.3.0.0(${beginTime}, ${endTime}, ${environmentType})`;
  isProd ? logging(version) : console.log(version);
  const isoDate = new Date().toLocaleDateString('en-CA');
  if (beginTime.indexOf('T') < 0) beginTime = isoDate + 'T' + beginTime;
  if (endTime.indexOf('T') < 0) endTime = isoDate + 'T' + endTime;
  isProd ? console.plog(beginTime, endTime) : console.log(beginTime, endTime);
  if (Date.now() < Date.parse(beginTime)) return;
  if (Date.now() > Date.parse(endTime)) return;
  const nowEnvironmentType = isProd
    ? MhUtils.getEnvironmentType()
    : user.environment_type;
  if (nowEnvironmentType === environmentType) return;
  isProd
    ? console.plog(environmentType, beginTime, endTime, callback)
    : console.log(environmentType, beginTime, endTime, callback);
  isProd
    ? MhUtils.travel(environmentType, timeout, callback)
    : travelMock(environmentType, timeout, callback);
}
function travelMock(environmentType, milliseconds, callback) {
  const MOUSEHUNT_BASE_URL = 'https://www.mousehuntgame.com/';
  if (user.environment_type === environmentType) {
    if (callback) callback();
    return;
  }
  const timeout = milliseconds ? milliseconds : 2000;
  hg.utils.User.travel(environmentType, () => {
    if (timeout >= 0) {
      window.setTimeout(function () {
        console.log(callback);
        if (callback) callback();
        else window.location.href = MOUSEHUNT_BASE_URL;
      }, timeout);
    }
  });
}
