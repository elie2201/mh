function furomaAssignment() {
  const isProd = false;
  const eventLocation = localStorage['eventLocation'];
  if (eventLocation !== 'library_assignment') return;
  logging('furomaAssignment_0.0.1.0.5');
  // my own hard-coded constants
  /** localStorage settings */
  const defaultSettings = {
    traps: {
      default: ['Gouda Cheese', 'best.weapon.tactical', 'Seasonal Gift', 'None']
    }
  };
  const settings = defaultSettings;
  // MH constants
  const quest =
    user.quests.QuestFuromaResearch ||
    user.quests.QuestPagodaResearch ||
    user.quests.QuestAdvancedPagodaResearch;
  const furomaAssignmentUtility = (() => {
    const utils = {
      // values
      /** ajax URI */
      URI: 'managers/ajax/users/questsprogress.php',
      /** Set immutable fields */
      setImmutable() {
        const immutableFields = ['URI'];
        for (let i = 0; i < immutableFields.length; i++) {
          const immutableField = immutableFields[i];
          Object.defineProperty(utils, immutableField, {
            // value: 'This cannot be changed',
            writable: false, // Makes the field immutable
            configurable: false, // Prevents deletion or re-configuration of the field
            enumerable: true // Makes the field visible during enumeration (e.g., in a for...in loop)
          });
        }
      },
      /** trapState, default: xxx */
      trapState: 'default',
      environmentType: 'dojo',
      bait: '',
      weapon: '',
      base: '',
      trinket: '',
      /** start running */
      run() {
        logging('start furomaAssignment');
        this.autoSettingsPromise()
          .then(this.checkAssignmentPromise)
          .then(this.handleTrapPromise);
      },
      /** auto settings */
      autoSettingsPromise() {
        logging('start autoSettingsPromise');
        return new Promise((resolve, reject) => {
          // auto settings for running if necessary and then invoke next promise
          resolve();
        });
      },
      /**
       * Check library assignment.
       * Smash research quest item if assignment contains following keywords:
       * Zugzwang, Onyx Mallet, Maki Cheese.
       */
      checkAssignmentPromise() {
        logging(`start checkAssignmentPromise`);
        return new Promise((resolve, reject) => {
          console.log(`eventLocation: ${eventLocation}, quest:`, quest);
          if (eventLocation !== 'library_assignment' || !quest) {
            console.log('Have no assignment,', resolve);
            if (resolve) resolve();
            return;
          }
          console.log('fetch assignment data and parse progress', resolve);
          utils
            .fetchLibraryAssignmentDataPromise()
            .then(utils.parseProgressPromise)
            .then(() => {
              if (resolve) resolve();
            });
        });
      },
      /**
       * A Promise wrap of ajax query assignment progress.
       * Utility-like, without any logic inside.
       */
      fetchLibraryAssignmentDataPromise() {
        logging(`start fetchLibraryAssignmentDataPromise`);
        return new Promise((resolve, reject) => {
          this.ajax(null, {}, resolve);
        });
      },
      parseProgressPromise(assignmentData) {
        logging(`start parseProgressPromise(`, assignmentData);
        return new Promise((resolve, reject) => {
          const progress = assignmentData['progress'];
          let assignment = null;
          for (const key in progress) {
            if (!Object.hasOwn(progress, key)) continue;
            assignment = progress[key];
          }
          const tasks = assignment['progress'];
          for (let i = 0; i < tasks.length; i++) {
            const task = tasks[i];
            const percentage = parseInt(task['percentage']);
            if (percentage === 100) continue;
            const content = task['string'];
            utils.parseContent(content);
            break;
          }
          tasks.forEach((v, i) => {
            if (parseInt(v['percentage']) !== 100)
              if (
                v['string'].indexOf('Power Charm') > -1 &&
                utils.environmentType === 'dojo'
              )
                utils.trinket = 'Power Charm';
              else if (
                v['string'].indexOf('Luck Charm') > -1 &&
                utils.environmentType === 'dojo'
              )
                utils.trinket = 'Luck Charm';
          });
          console.log(utils);
          if (resolve) resolve();
        });
      },
      parseContent(string) {
        if (string.indexOf('Hapless') > -1) {
          this.environmentType = 'meditation_room';
          this.bait = 'Brie Cheese';
        } else if (string.indexOf('Master of the Dojo') > -1) {
          this.environmentType = 'pinnacle_chamber';
        } else if (string.indexOf('Masters of the Cheese Belt') > -1) {
          this.environmentType = 'meditation_room';
          this.bait = 'Glutter Cheese';
        } else if (string.indexOf('Masters of the Cheese Fang') > -1) {
          this.environmentType = 'meditation_room';
          this.bait = 'Combat Cheese';
        } else if (string.indexOf('Masters of the Cheese Claw') > -1) {
          this.environmentType = 'meditation_room';
          this.bait = 'Susheese Cheese';
        }
        if (string.indexOf('Training Grounds') > -1)
          this.environmentType = 'training_grounds';
        if (string.indexOf('in the Dojo') > -1) this.environmentType = 'dojo';
        if (string.indexOf('Bamboozler Base') > -1)
          this.base = 'Bamboozler Base';
        if (string.indexOf('Power Charm') > -1) this.trinket = 'Power Charm';
        if (string.indexOf('Luck Charm') > -1) this.trinket = 'Luck Charm';
      },
      /** handle trap promise */
      handleTrapPromise() {
        logging('start handleTrapPromise');
        console.log('environmentType:', utils.environmentType);
        return new Promise((resolve, reject) => {
          travel(utils.environmentType, 2000, () => {
            const trap = settings.traps[utils.trapState];
            if (utils.bait !== '') trap[0] = utils.bait;
            if (utils.weapon !== '') trap[1] = utils.weapon;
            if (utils.base !== '') trap[2] = utils.base;
            if (utils.trinket !== '') trap[3] = utils.trinket;
            if (isProd) armAllTraps(...trap);
            else console.log(...trap);
            if (resolve) resolve();
          });
        });
      },
      /** location ajax */
      ajax(element, params, successCallback, errorCallback) {
        params['last_read_journal_entry_id'] = $('#journallatestentry').data(
          'entry-id'
        );
        params['uh'] = user.unique_hash;
        $.ajax({
          data: params,
          dataType: 'json',
          type: 'post',
          url: MOUSEHUNT_BASE_URL + this.URI,
          success: function (data) {
            eventRegistry.doEvent('ajax_response', data);
            if (successCallback) {
              successCallback(data);
            }
          },
          // prettier-ignore
          error: function (data) {
            // eslint-disable-next-line new-cap
            const popup = new jsDialog();
            popup.setTemplate('error');
            popup.addToken('{*title*}', 'An error has occurred.');
            popup.addToken('{*content*}', 'An error occurred. Please try again!');
            popup.show();
            if (errorCallback) errorCallback(data);
          }
        });
      }
    };
    // initial utils
    function init() {
      // make fields immutable
      utils.setImmutable();
      // initialize values in utils if necessary
    }
    init();
    return utils;
  })();
  furomaAssignmentUtility.run();
  function travel(environmentType, milliseconds, callback) {
    if (
      isProd
        ? MhUtils.getEnvironmentType() === environmentType
        : user.environment_type === environmentType
    ) {
      if (callback) callback();
      return;
    }
    const timeout = milliseconds ? milliseconds : 2000;
    hg.utils.User.travel(environmentType, () => {
      if (timeout >= 0) {
        window.setTimeout(function () {
          if (callback) callback();
          else MhUtils.reloadWithMessage('After travel. Reloading...', false);
        }, timeout);
      }
    });
  }
}
