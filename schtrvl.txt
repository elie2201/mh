// #scheduledTravel
// 1.爬塔避開爛時段靈活版
// 2.好時段刷 GP(提前 1.5小時進去,讓好時段時已經在百科全書)
(() => {
  const storageKey = 'allLuckyUnlucky';
  const settings = MyUtils.getStorageToObject(storageKey, {});
  const setting = settings[user.firstname];
  const getPlan = (i, j, k) => {
    return {
      beginTime: setting[i],
      endTime: setting[j],
      environmentType: setting[k]
    };
  };
  const leavePlans = [];
  const backPlans = [];
  const maxIndex = Math.floor(setting.length / 3);
  for (let i = 0; i < maxIndex; i++) {
    const beginIndex = i * 3;
    const go = setting[beginIndex + 2];
    if (go)
      leavePlans.push(getPlan(beginIndex, beginIndex + 1, beginIndex + 2));
    const back = setting[beginIndex + 5];
    if (back) {
      backPlans.push(getPlan(beginIndex + 3, beginIndex + 4, beginIndex + 5));
      i++;
    }
  }
  console.debug(leavePlans, backPlans);
  if (
    backPlans?.[0]?.environmentType !== 'rift_valour' ||
    user?.quests?.QuestRiftValour?.state.trim().toLowerCase() === 'tower'
  )
    if (mh.addon.TravelPlan.run(leavePlans)) throw new Error('Has traveled');
  if (backPlans.length > 0)
    if (mh.addon.TravelPlan.run(backPlans)) throw new Error('Has traveled');
})();
// End #scheduledTravel
