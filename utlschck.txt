/** simply run check() and then run act() if check() return true */
function checkAndAct(checks) {
  // const isProd = false;
  logging('checkAndAct_0.0.0.0.2', checks);
  for (let i = 0; i < checks.length; i++) {
    const check = checks[i];
    if (getEnvironmentType() !== check.atEnvironmentType) continue;
    console.log('run check');
    if (check.check()) {
      if (check.act) {
        check.act();
        return;
      }
    }
  }
}
